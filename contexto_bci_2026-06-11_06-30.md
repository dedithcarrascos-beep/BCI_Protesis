M_pwr_env = filter(num_alpha, den_alpha, env_log_M);

% ---------------------------------------------------------

% Calcular Media y Desviación Estándar durante la calibración inicial

t_resp_tr = t_resp_tr; % tiempo estimado respuesta transitoria

idx_cal_fin = round(tiempo_inicio * fs - n_actual - n_gap);
idx_cal_ini = idx_cal_fin - n_base + 1;

C3_reposo_prom_mu = mean(C3_pwr_env_mu(idx_cal_ini : idx_cal_fin));

C43_reposo_std_mu  = std(C43_pwr_env_mu(idx_cal_ini : idx_cal_fin));

C3_reposo_prom_beta = mean(C3_pwr_env_beta(idx_cal_ini : idx_cal_fin));

C4_reposo_std_beta  = std(C4_pwr_env_beta(idx_cal_ini : idx_cal_fin));

% ---------------------------------------------------------
% NUEVO - baseline fija
% ---------------------------------------------------------
idx_rest = (t_filt >= 1 & t_filt < 9) | (t_filt >= 21 & t_filt < 29);
C3_reposo_prom_mu   = mean(C3_pwr_env_mu(idx_rest));

% ---------------------------------------------------------

M_reposo_prom = mean(M_pwr_env(idx_cal_ini : idx_cal_fin)); % 2.5 para estabilizar resp transitoria
M_reposo_std  = std(M_pwr_env(idx_cal_ini : idx_cal_fin));

% Definir % ERD y % ERS
% k_erd_mu = 0.20; % caida del 20%
% k_erd_beta = 0.15; % caida del 15%
% k_ers_beta = 0.20; % rebote del 20%
% k_emg = 0.20; % 20% más energía = movimiento voluntario
k_erd_mu = k_erd_mu; % caida del 20%
k_erd_beta = k_erd_beta; % caida del 15%
k_ers_beta = k_ers_beta; % rebote del 20%
k_emg = k_emg; % 20% más energía = movimiento voluntario

ERD_C3_mu_umbral = log(1-k_erd_mu); % signo - es caída
ERD_C4_mu_umbral = log(1-k_erd_mu);
ERD_C43_mu_umbral = log(1-k_erd_mu);
ERD_C3_beta_umbral = log(1-k_erd_beta);
ERD_C4_beta_umbral = log(1-k_erd_beta);
ERD_C43_beta_umbral = log(1-k_erd_beta);
ERS_C3_beta_umbral = log(1+k_ers_beta); % sigo + es rebote
ERS_C4_beta_umbral = log(1+k_ers_beta);
ERS_C43_beta_umbral = log(1+k_ers_beta);

% umbral_correlacion = 0.6; % Arriba del 60% de similitud con el EMG = Artefacto
umbral_correlacion = umbral_correlacion; % Arriba del 60% de similitud con el EMG = Artefacto
Umbral_Ruido_Extremo = Umbral_Ruido_Extremo; % picos que superan el 200% 

M_umbral = log(1+k_emg) + M_reposo_prom; % 20 más energía
%M_umbral = (1 + k_emg) * M_reposo_prom;
M_umbral_on  = M_reposo_prom + (k_emg_on * M_reposo_std);
M_umbral_off = M_reposo_prom + (k_emg_off * M_reposo_std);

muestras_totales = length(C3_pwr_env_mu);

% Parámetros de la ventana en MUESTRAS (fs = 250)
% win_n = fs * 1;             % Ventana de 1 segundo (250 muestras)
% tiempo_inicio = t_calibracion + 0.5 + 1; % 5 calibración + 0.5 filtro promedio + 1 ventana
% step_n = round(fs * 0.25);  % Desplazamiento de 0.25 seg (63 muestras)
%win_n = win_n;             % Ventana de 1 segundo (250 muestras)
tiempo_inicio = tiempo_inicio; % 5 calibración + 0.5 filtro promedio + 1 ventana
%step_n = step_n;  % Desplazamiento de 0.25 seg (63 muestras)

% Vectores de salida (inicializados en 0)
C3_erd_t_mu = zeros(muestras_totales, 1);
C4_erd_t_mu = zeros(muestras_totales, 1);
C43_erd_t_mu = zeros(muestras_totales, 1);
C3_mu_t  = zeros(muestras_totales, 1);
C4_mu_t  = zeros(muestras_totales, 1);
C43_mu_t  = zeros(muestras_totales, 1);
C3_frec_t_mu = zeros(muestras_totales, 1);
C4_frec_t_mu = zeros(muestras_totales, 1);
C43_frec_t_mu = zeros(muestras_totales, 1);

C3_erd_t_beta = zeros(muestras_totales, 1);
C4_erd_t_beta = zeros(muestras_totales, 1);
C43_erd_t_beta = zeros(muestras_totales, 1);
C3_beta_t  = zeros(muestras_totales, 1);
C4_beta_t  = zeros(muestras_totales, 1);
C43_beta_t  = zeros(muestras_totales, 1);
C3_frec_t_beta = zeros(muestras_totales, 1);
C4_frec_t_beta = zeros(muestras_totales, 1);
C43_frec_t_beta = zeros(muestras_totales, 1);

EMG_pulso_t = zeros(muestras_totales, 1);

C3_motor_t = zeros(muestras_totales, 1);
C4_motor_t = zeros(muestras_totales, 1);
C43_motor_t = zeros(muestras_totales, 1);

% Arreglos para guardar la historia de la línea base y umbrales en el tiempo
Base_C3_mu_t = zeros(muestras_totales, 1);
Base_C4_mu_t = zeros(muestras_totales, 1);
Base_C43_mu_t = zeros(muestras_totales, 1);

Base_C3_beta_t = zeros(muestras_totales, 1);
Base_C4_beta_t = zeros(muestras_totales, 1);
Base_C43_beta_t = zeros(muestras_totales, 1);

% Nuevos arreglos para guardar el estado de los artefactos
C3_artefacto_t = zeros(muestras_totales, 1);
C4_artefacto_t = zeros(muestras_totales, 1);
C43_artefacto_t = zeros(muestras_totales, 1);
artefacto_c3 = 0; artefacto_c4 = 0; artefacto_c43 = 0;

% Variables retenedoras (Zero-order hold)
erd_c3_actual_mu = 0; erd_c4_actual_mu = 0; erd_c43_actual_mu = 0;
est_c3_actual_mu = 0; est_c4_actual_mu = 0; est_c43_actual_mu = 0;
frec_mov_c3_mu = 0; frec_mov_c4_mu = 0; frec_mov_c43_mu = 0;

erd_c3_actual_beta = 0; erd_c4_actual_beta = 0; erd_c43_actual_beta = 0;
est_c3_actual_beta = 0; est_c4_actual_beta = 0; est_c43_actual_beta = 0;
frec_mov_c3_beta = 0; frec_mov_c4_beta = 0; frec_mov_c43_beta = 0;

estado_mov_emg = 0; % 0 = reposo, 1 = movimiento activo

comando_motor_c3 = 0; comando_motor_c4 = 0; comando_motor_c43 = 0;
contador_c3 = 0; contador_c4 = 0; contador_c43 = 0;

timer_c3 = 0; timer_c4 = 0; 
max_ciclos_on = round(1.0 / t_step); % 1.0 seg / 0.25 seg = 4 ciclos máximos

% Temporizadores de bloqueo para proteger la línea base
bloqueo_base_c3 = 0; bloqueo_base_c4 = 0; bloqueo_base_c43 = 0;
% Tiempo = Ventana Actual + Gap + Ventana Base
muestras_bloqueo = n_actual + n_gap + n_base;

% Arreglos para el semáforo del Frontend
C3_led_rojo_t = zeros(muestras_totales, 1);
C4_led_rojo_t = zeros(muestras_totales, 1);

% Variables de estado actual
led_rojo_c3 = 0; 
led_rojo_c4 = 0;

flanco_subida = 0;
ev=[];

% El bucle inicia a partir de que se llena la primera ventana de 1 segundo
%for i = win_n + tiempo_inicio * fs : muestras_totales
for i = round(tiempo_inicio * fs) : muestras_totales
    
    % ¿Es momento de actualizar los cálculos? (Se ejecuta solo cada ventana actual)
    if mod(i, step_n) == 0
    %if mod(i, n_actual) == 0

        rango_actual = (i - n_actual + 1) : i;
        fin_base   = i - n_actual - n_gap;
        rango_base = (fin_base - n_base + 1) : fin_base;
        
        % 1. Extraer la ventana actual de 1 segundo (Nota: sin el doble log)
        prom_mov_c3_mu   = mean(C3_pwr_env_mu(rango_actual));
        prom_mov_c4_mu   = mean(C4_pwr_env_mu(rango_actual));
        prom_mov_c43_mu  = mean(C43_pwr_env_mu(rango_actual));
        prom_mov_c3_beta = mean(C3_pwr_env_beta(rango_actual));
        prom_mov_c4_beta = mean(C4_pwr_env_beta(rango_actual));
        prom_mov_c43_beta= mean(C43_pwr_env_beta(rango_actual));
        
        % Extraer ventana del EMG para la correlación
        %win_m = M_pwr_env(rango);
        win_m = M_pwr_env(rango_actual);
        %win_m = env_log_M(rango_actual);
        prom_mov_m = mean(win_m); % Promedio de la ventana actual del EMG

        if prom_mov_m > M_umbral_on
            estado_mov_emg = 1; % Sube el pulso
            if flanco_subida == 0
                flanco_subida = 1;
                t_emg_on = i / fs;
            end
        elseif prom_mov_m < M_umbral_off
            estado_mov_emg = 0; % Baja el pulso solo si el músculo se relajó de verdad
            if flanco_subida == 1
                flanco_subida = 0;
                t_emg_off = i / fs;
                ev = [ev; [t_emg_on t_emg_off]];
            end
        end

        % ---------------------------------------------------------
        % 6.1 DETECCIÓN DE ARTEFACTOS POR CORRELACIÓN (Pearson)
        % ---------------------------------------------------------
        win_m_z = win_m - mean(win_m);
        
        % Actualizar temporizadores de bloqueo (restando el step_n en cada ciclo)
        bloqueo_base_c3 = max(0, bloqueo_base_c3 - step_n);
        bloqueo_base_c4 = max(0, bloqueo_base_c4 - step_n);
        bloqueo_base_c43 = max(0, bloqueo_base_c43 - step_n);

        % Para C3
        win_c3_z = env_log_C3_mu(rango_actual) - mean(env_log_C3_mu(rango_actual));
        corr_c3 = sum(win_c3_z .* win_m_z) / (sqrt(sum(win_c3_z.^2)) * sqrt(sum(win_m_z.^2)));
        if isnan(corr_c3), corr_c3 = 0; end
        artefacto_c3 = abs(corr_c3) > umbral_correlacion;
        
        % Si hay artefacto HOY, congelamos la base hacia el FUTURO
        if artefacto_c3 == 1
            bloqueo_base_c3 = muestras_bloqueo; 
        end

        % Para C4
        win_c4_z = env_log_C4_mu(rango_actual) - mean(env_log_C4_mu(rango_actual));
        corr_c4 = sum(win_c4_z .* win_m_z) / (sqrt(sum(win_c4_z.^2)) * sqrt(sum(win_m_z.^2)));
        if isnan(corr_c4), corr_c4 = 0; end
        artefacto_c4 = abs(corr_c4) > umbral_correlacion;
        
        if artefacto_c4 == 1
            bloqueo_base_c4 = muestras_bloqueo;
        end

        % Para C4 - C3
        win_c43_z = env_log_C43_mu(rango_actual) - mean(env_log_C43_mu(rango_actual));
        corr_c43 = sum(win_c43_z .* win_m_z) / (sqrt(sum(win_c43_z.^2)) * sqrt(sum(win_m_z.^2)));
        if isnan(corr_c43), corr_c43 = 0; end
        artefacto_c43 = abs(corr_c43) > umbral_correlacion;
        
        if artefacto_c43 == 1
            bloqueo_base_c43 = muestras_bloqueo;
        end

        % ---------------------------------------------------------
        % 6.2 CALCULAR ERD (POTENCIA LOGARÍTMICA)
        % ---------------------------------------------------------
        % Al ser logaritmos, la resta directa indica la proporción de cambio
        erd_c3_actual_mu = prom_mov_c3_mu - C3_reposo_prom_mu;
        erd_c4_actual_mu = prom_mov_c4_mu - C4_reposo_prom_mu;
        erd_c43_actual_mu = prom_mov_c43_mu - C43_reposo_prom_mu;

        erd_c3_actual_beta = prom_mov_c3_beta - C3_reposo_prom_beta;
        erd_c4_actual_beta = prom_mov_c4_beta - C4_reposo_prom_beta;
        erd_c43_actual_beta = prom_mov_c43_beta - C43_reposo_prom_beta;

        % ---------------------------------------------------------
        % 6.3 DETECCIÓN DE ARTEFACTOS POR PICOS GIGANTES
        % ---------------------------------------------------------        

        % Umbral de ruido extremo (ejemplo: si la potencia sube 200% sobre la base)
        Umbral_Ruido_Extremo = Umbral_Ruido_Extremo; 
        
        % Si AMBAS bandas experimentan una explosión de energía hacia ARRIBA al mismo tiempo:
        if (erd_c3_actual_mu > Umbral_Ruido_Extremo) && (erd_c3_actual_beta > Umbral_Ruido_Extremo)
            artefacto_c3 = 1; % ¡Es un jalón de cable o un parpadeo gigante!
            bloqueo_base_c3 = muestras_bloqueo;
        end

        % Si AMBAS bandas experimentan una explosión de energía hacia ARRIBA al mismo tiempo:
        if (erd_c4_actual_mu > Umbral_Ruido_Extremo) && (erd_c4_actual_beta > Umbral_Ruido_Extremo)
            artefacto_c4 = 1; % ¡Es un jalón de cable o un parpadeo gigante!
            bloqueo_base_c4 = muestras_bloqueo;
        end

        % Si AMBAS bandas experimentan una explosión de energía hacia ARRIBA al mismo tiempo:
        if (erd_c43_actual_mu > Umbral_Ruido_Extremo) && (erd_c43_actual_beta > Umbral_Ruido_Extremo)
            artefacto_c43 = 1; % ¡Es un jalón de cable o un parpadeo gigante!
            bloqueo_base_c43 = muestras_bloqueo;
        end

        % ---------------------------------------------------------
        % SEMÁFORO PARA EL FRONTEND (Lógica UI)
        % ---------------------------------------------------------
        % Verdadero (1) = No hacer movimiento (Artefacto o Base congelada)
        % Falso (0)   = Luz verde para mover
        
        led_rojo_c3 = (artefacto_c3 == 1) || (bloqueo_base_c3 > 0);
        led_rojo_c4 = (artefacto_c4 == 1) || (bloqueo_base_c4 > 0);

        % ---------------------------------------------------------
        % 6.4 LÍNEA BASE DINÁMICA (ÚLTIMOS X SEGUNDOS)
        % ---------------------------------------------------------
                
        % Actualizar reposo C3 SOLO si: Motor apagado Y el bloqueo expiró
        if (comando_motor_c3 == 0) && (bloqueo_base_c3 == 0)
            C3_reposo_prom_mu = mean(C3_pwr_env_mu(rango_base));
            C3_reposo_prom_beta = mean(C3_pwr_env_beta(rango_base));
        end
        
        % Actualizar reposo C4 SOLO si: Motor apagado Y el bloqueo expiró
        if (comando_motor_c4 == 0) && (bloqueo_base_c4 == 0)
            C4_reposo_prom_mu = mean(C4_pwr_env_mu(rango_base));
            C4_reposo_prom_beta = mean(C4_pwr_env_beta(rango_base));
        end

        % Actualizar reposo C4-C3 SOLO si: Motor apagado Y en ESTE instante no hay artefacto
        if (comando_motor_c43 == 0) && (bloqueo_base_c43 == 0)
            C43_reposo_prom_mu = mean(C43_pwr_env_mu(rango_base));
            C43_reposo_prom_beta = mean(C43_pwr_env_beta(rango_base));
        end
        %end

        % ---------------------------------------------------------
        % NUEVO - baseline fija
        % ---------------------------------------------------------
        idx_rest = (t_filt >= 1 & t_filt < 9) | (t_filt >= 21 & t_filt < 29);
        C3_reposo_prom_mu   = mean(C3_pwr_env_mu(idx_rest));
        C3_reposo_prom_beta = mean(C3_pwr_env_beta(idx_rest));
        C4_reposo_prom_mu   = mean(C4_pwr_env_mu(idx_rest));
        C4_reposo_prom_beta = mean(C4_pwr_env_beta(idx_rest));
        % ---------------------------------------------------------
       
        % ---------------------------------------------------------
        % 6.5 MÁQUINA DE ESTADOS COMBINADA (CON TIMEOUT Y ERS)
        % ---------------------------------------------------------
        
        % Hemisferio Izquierdo (C3) -> Controla Motor Derecho
        if comando_motor_c3 == 1
            % EL MOTOR ESTÁ ENCENDIDO: Evaluamos cuándo apagarlo
            timer_c3 = timer_c3 + 1;
            
            % Apagar si: pasó 1 segundo (timeout), hay rebote ERS, o surge un artefacto
            if (timer_c3 >= max_ciclos_on) || (erd_c3_actual_beta > ERS_C3_beta_umbral) || (artefacto_c3 == 1)
                comando_motor_c3 = 0;
                contador_c3 = 0; % Reset para exigir nueva intención
                timer_c3 = 0;
            end
            
        else
            % EL MOTOR ESTÁ APAGADO: Evaluamos si debe encenderse
            %if (erd_c3_actual_mu < ERD_C3_mu_umbral) && (erd_c3_actual_beta < ERD_C3_beta_umbral) && (artefacto_c3 == 0)
            if (erd_c3_actual_mu < ERD_C3_mu_umbral) && (artefacto_c3 == 0)
                contador_c3 = contador_c3 + 1;
                
                if contador_c3 >= K_persistencia
                    comando_motor_c3 = 1; % ¡Encendido!
                    timer_c3 = 0;         % Inicia el cronómetro de 1 segundo
                end
            elseif (erd_c3_actual_beta > ERS_C3_beta_umbral) || (artefacto_c3 == 1)
                contador_c3 = 0; % Resetea la intención si hay rebote o ruido
            end
        end
        
        
        % Hemisferio Derecho (C4) -> Controla Motor Izquierdo
        if comando_motor_c4 == 1
            % EL MOTOR ESTÁ ENCENDIDO: Evaluamos cuándo apagarlo
            timer_c4 = timer_c4 + 1;
            
            if (timer_c4 >= max_ciclos_on) || (erd_c4_actual_beta > ERS_C4_beta_umbral) || (artefacto_c4 == 1)
                comando_motor_c4 = 0;
                contador_c4 = 0;
                timer_c4 = 0;
            end
            
        else
            % EL MOTOR ESTÁ APAGADO: Evaluamos si debe encenderse
            %if (erd_c4_actual_mu < ERD_C4_mu_umbral) && (erd_c4_actual_beta < ERD_C4_beta_umbral) && (artefacto_c4 == 0)
            if (erd_c4_actual_mu < ERD_C4_mu_umbral) && (artefacto_c4 == 0)
                contador_c4 = contador_c4 + 1;
                
                if contador_c4 >= K_persistencia
                    comando_motor_c4 = 1;
                    timer_c4 = 0;
                end
            elseif (erd_c4_actual_beta > ERS_C4_beta_umbral) || (artefacto_c4 == 1)
                contador_c4 = 0;
            end
        end
    end
    
    % Retenedor: Asignamos el valor actual a este instante de tiempo
    C3_erd_t_mu(i) = erd_c3_actual_mu;
    C4_erd_t_mu(i) = erd_c4_actual_mu;
    C43_erd_t_mu(i) = erd_c43_actual_mu;
    C3_mu_t(i)  = est_c3_actual_mu;
    C4_mu_t(i)  = est_c4_actual_mu;
    C43_mu_t(i)  = est_c43_actual_mu;

    C3_erd_t_beta(i) = erd_c3_actual_beta;
    C4_erd_t_beta(i) = erd_c4_actual_beta;
    C43_erd_t_beta(i) = erd_c43_actual_beta;
    C3_beta_t(i)  = est_c3_actual_beta;
    C4_beta_t(i)  = est_c4_actual_beta;
    C43_beta_t(i)  = est_c43_actual_beta;

    EMG_pulso_t(i) = estado_mov_emg;

    C3_motor_t(i) = comando_motor_c3;
    C4_motor_t(i) = comando_motor_c4;

    C3_artefacto_t(i) = artefacto_c3;
    C4_artefacto_t(i) = artefacto_c4;

    C3_led_rojo_t(i) = led_rojo_c3;
    C4_led_rojo_t(i) = led_rojo_c4;

    % Retenedor Zero-Order Hold para la línea base dinámica
    Base_C3_mu_t(i) = C3_reposo_prom_mu;
    Base_C4_mu_t(i) = C4_reposo_prom_mu;
    Base_C43_mu_t(i) = C43_reposo_prom_mu;
    
    Base_C3_beta_t(i) = C3_reposo_prom_beta;
    Base_C4_beta_t(i) = C4_reposo_prom_beta;
    Base_C43_beta_t(i) = C43_reposo_prom_beta;
end

          'CH5 (M1)', 'CH6 (M2)', 'CH7 (C3'''')', 'CH8 (C4'''')'};

% Filtrar pasabandas estrecho alrededor de 31.25 Hz

fprintf('-------------------------------------------------------\n');
fprintf('Canal           V_31Hz (µV)   Z (kΩ)    Calidad\n');

    x = signals(ch, :) - mean(signals(ch, :));
    x_filt = filtfilt(b, a, x);
    
    % Amplitud RMS de la componente 31.25 Hz
    V_rms_31Hz = std(x_filt);
    V_pp_31Hz = V_rms_31Hz * 2 * sqrt(2);
    
    % Impedancia: Vpp = 4 * I * Z  (asumiendo Z_INP ≈ Z_INN)
    %Z_kOhm = (V_pp_31Hz * 1e-6) / (4 * I_leadoff) / 1000;
    Z_kOhm = (V_pp_31Hz * 1e-6) / (2 * I_leadoff) / 1000;
    
    % Calificación
    if Z_kOhm < 5
        calidad = '✓ Excelente';
    elseif Z_kOhm < 10

    elseif Z_kOhm < 25

        calidad = '✗ Mala';
    else
        calidad = '✗✗ Desconectado/muy mala';
    end
    
    fprintf('%-15s %8.1f      %6.1f    %s\n', ...
            labels{ch}, V_pp_31Hz, Z_kOhm, calidad);

    error('Grabación demasiado corta. Se necesitan al menos 30 segundos.');

% =========================================================

[b_lp, a_lp] = butter(4, 40/(fs/2), 'low');
[b_mu, a_mu] = butter(4, [8 13]/(fs/2), 'bandpass');
[b_beta, a_beta] = butter(4, [15 30]/(fs/2), 'bandpass');

[b_emg, a_emg] = butter(4, [20 100]/(fs/2), 'bandpass');

% =========================================================

C3_clean   = preprocess_eeg(C3);
C3p_clean  = preprocess_eeg(C3p);
C3pp_clean = preprocess_eeg(C3pp);
C4_clean   = preprocess_eeg(C4);
C4p_clean  = preprocess_eeg(C4p);
C4pp_clean = preprocess_eeg(C4pp);

C3pp_proc = C3pp_clean;
C4p_proc  = C4p_clean;
C4pp_proc = C4pp_clean;

C4_lap_mu   = filtfilt(b_mu, a_mu, C4_lap);

C3pp_mu     = filtfilt(b_mu, a_mu, C3pp_proc);
C4p_mu      = filtfilt(b_mu, a_mu, C4p_proc);
C4pp_mu     = filtfilt(b_mu, a_mu, C4pp_proc);

/*
 * FIRMWARE UNIFICADO BCI: ADS1299 -> SERIAL (USB)
 *
 * Combina lead-off (latiguillos) + adquisición normal (protocolos) + (PASO 10) motores.
 * El PC (Go/Wails) selecciona el modo enviando comandos de 2 bytes [CMD][ARG].
 *
 * Ramón Felipe Miranda Hernández
 * 2026 - junio - 02
 *
 * Datos ESP32 -> PC: header 0xA0 0xC0 + 8 canales x4 bytes (int32), 921600 baud.
 * Comandos  PC -> ESP32 (2 bytes fijos):
 *   'I',0x00  -> IDLE      : detener stream (servos a 150° - PASO 10)
 *   'L',0x00  -> LEAD-OFF  : initADS1299_leadoff() + stream
 *   'T',0x00  -> NORMAL    : initADS1299_normal() + stream (entrenamiento)
 *   'R',0x00  -> REALTIME  : initADS1299_normal() + stream + motores (PASO 10)
 *   'M',state -> MOTORES   : solo en REALTIME, bit0=C3, bit1=C4 (1=100°, 0=150°) (PASO 10)
 */

#include <SPI.h>

// --- PINOUT ---
const int PIN_CS = 5;
const int PIN_DRDY = 4;
// NOTA: El pin START del ADS1299 debe estar conectado físicamente a 3.3V
// SPI: 18 (CLK), 19 (MISO), 23 (MOSI)

// --- Servos MG90S (control de prótesis) ---
const int PIN_SERVO_C3 = 25;  // motor C3
const int PIN_SERVO_C4 = 26;  // motor C4
const int CH_SERVO_C3  = 4;   // canal LEDC (0-15; SPI no usa LEDC)
const int CH_SERVO_C4  = 5;
const int SERVO_FREQ   = 50;  // 50 Hz (periodo 20 ms) para MG90S
const int SERVO_RES    = 16;  // resolución PWM en bits
const int ANG_REPOSO   = 150; // posición inicial (señal = 0)
const int ANG_ACTIVO   = 100; // prótesis movida   (señal = 1)

// --- Comandos ADS1299 ---
const byte WREG   = 0x40;
const byte RDATAC = 0x10;
const byte SDATAC = 0x11;
const byte RESET  = 0x06;

// --- Modos de operación ---
enum Modo { MODO_IDLE, MODO_LEADOFF, MODO_NORMAL, MODO_REALTIME };
Modo modo = MODO_IDLE;

// --- Estructura de Datos (Buffer Crudo) ---
const int PACKET_SIZE = 34;
uint8_t eegPayload[PACKET_SIZE]; // 2 cabecera + 32 datos
long channelData[8];

void setup() {
  Serial.begin(921600);

  eegPayload[0] = 0xA0;
  eegPayload[1] = 0xC0;

  pinMode(PIN_CS, OUTPUT);
  pinMode(PIN_DRDY, INPUT);
  digitalWrite(PIN_CS, HIGH);

  SPI.begin(18, 19, 23, 5);
  SPI.beginTransaction(SPISettings(2000000, MSBFIRST, SPI_MODE1));
  delay(500); // Esperar arranque del ADS1299

  // Inicializar servos (PWM LEDC) y dejarlos en reposo (150°).
  ledcSetup(CH_SERVO_C3, SERVO_FREQ, SERVO_RES);
  ledcAttachPin(PIN_SERVO_C3, CH_SERVO_C3);
  ledcSetup(CH_SERVO_C4, SERVO_FREQ, SERVO_RES);
  ledcAttachPin(PIN_SERVO_C4, CH_SERVO_C4);
  setMotores(0x00); // ambos a 150°

  // Arranca en IDLE: no transmite hasta recibir un comando de modo desde el PC.
  modo = MODO_IDLE;
}

void loop() {
  procesarComandos();

  // El chip baja DRDY cuando tiene datos nuevos. Solo transmitimos si hay un modo activo.
  if (modo != MODO_IDLE && digitalRead(PIN_DRDY) == LOW) {
    readAndSendData();
  }
}

// -----------------------------------------------------------------------
// Parser de comandos PC -> ESP32 (mensajes fijos de 2 bytes [CMD][ARG])
// -----------------------------------------------------------------------
void procesarComandos() {
  while (Serial.available() >= 2) {
    byte cmd = Serial.read();
    byte arg = Serial.read();

    switch (cmd) {
      case 'I':  entrarIdle();              break;
      case 'L':  entrarLeadoff();           break;
      case 'T':  entrarNormal(MODO_NORMAL); break;
      case 'R':  entrarNormal(MODO_REALTIME); break;
      case 'M':  if (modo == MODO_REALTIME) setMotores(arg); break;
      default:   /* byte desconocido: ignorar */   break;
    }
  }
}

// -----------------------------------------------------------------------
// Transiciones de modo
// -----------------------------------------------------------------------
void entrarIdle() {
  modo = MODO_IDLE;
  sendCommand(SDATAC); // detener conversión continua
  setMotores(0x00);    // servos a 150° (reposo)
}

void entrarLeadoff() {
  initADS1299_leadoff();
  modo = MODO_LEADOFF;
}

void entrarNormal(Modo m) {
  initADS1299_normal();
  if (m == MODO_REALTIME) {
    setMotores(0x00); // arrancar con prótesis en reposo
  }
  modo = m;
}

// -----------------------------------------------------------------------
// Control de servos MG90S (PWM por LEDC)
// -----------------------------------------------------------------------

// anguloAUs convierte un ángulo (0-180°) a ancho de pulso en microsegundos
// (500 us = 0°, 2500 us = 180°), rango típico de servos hobby.
int anguloAUs(int ang) {
  return 500 + (int)((long)ang * 2000 / 180);
}

// usToDuty convierte microsegundos de pulso a valor de duty para LEDC
// (periodo de 20000 us a 50 Hz, resolución SERVO_RES bits).
uint32_t usToDuty(int us) {
  return (uint32_t)((uint64_t)us * ((1UL << SERVO_RES) - 1) / 20000UL);
}

void setServo(int canal, bool activo) {
  int ang = activo ? ANG_ACTIVO : ANG_REPOSO;
  ledcWrite(canal, usToDuty(anguloAUs(ang)));
}

// setMotores aplica el estado: bit0 = C3, bit1 = C4 (1 = 100°, 0 = 150°).
void setMotores(byte state) {
  setServo(CH_SERVO_C3, state & 0x01);
  setServo(CH_SERVO_C4, state & 0x02);
}

// -----------------------------------------------------------------------
// Configuración ADS1299 — modo NORMAL (sin inyección de corriente)
// -----------------------------------------------------------------------
void initADS1299_normal() {
  sendCommand(SDATAC);
  delay(10);

  // RESET por software: estado conocido
  sendCommand(RESET);
  delay(100);
  sendCommand(SDATAC);
  delay(10);

  writeRegister(0x01, 0x96);  // CONFIG1: 250 SPS
  writeRegister(0x02, 0xC0);  // CONFIG2: señal real
  writeRegister(0x03, 0xEC);  // CONFIG3: BIAS y ref interna ON

  // Blindaje anti-residuos: lead-off apagado
  writeRegister(0x04, 0x00);  // LOFF: sin inyección
  writeRegister(0x0F, 0x00);  // LOFF_SENSP: apagado
  writeRegister(0x10, 0x00);  // LOFF_SENSN: apagado
  writeRegister(0x17, 0x00);  // CONFIG4: comparadores apagados

  for (int i = 0x05; i <= 0x0C; i++) {
    writeRegister(i, 0x60);    // Canales con ganancia 24
  }

  writeRegister(0x0D, 0xCF);  // BIAS_SENSP (excluye EMG)
  writeRegister(0x0E, 0xCF);  // BIAS_SENSN (excluye EMG)
  writeRegister(0x15, 0x20);  // MISC1: SRB1 activo

  delay(50);
  sendCommand(RDATAC);
}

// -----------------------------------------------------------------------
// Configuración ADS1299 — modo LEAD-OFF (inyección 6 nA AC a 31.25 Hz)
// -----------------------------------------------------------------------
void initADS1299_leadoff() {
  sendCommand(SDATAC);
  delay(10);

  // RESET por software: estado conocido
  sendCommand(RESET);
  delay(100);
  sendCommand(SDATAC);
  delay(10);

  writeRegister(0x01, 0x96);  // CONFIG1: 250 SPS
  writeRegister(0x02, 0xC0);  // CONFIG2: señal real
  writeRegister(0x03, 0xEC);  // CONFIG3: BIAS y ref interna ON

  // LOFF: 6 nA AC a 31.25 Hz
  writeRegister(0x04, 0x02);

  for (int i = 0x05; i <= 0x0C; i++) {
    writeRegister(i, 0x60);    // Canales con ganancia 24
  }

  writeRegister(0x0D, 0xCF);  // BIAS_SENSP (excluye EMG)
  writeRegister(0x0E, 0xCF);  // BIAS_SENSN (excluye EMG)

  // Inyectar corriente SOLO en pines positivos
  writeRegister(0x0F, 0xFF);  // LOFF_SENSP activo
  writeRegister(0x10, 0x00);  // LOFF_SENSN apagado

  writeRegister(0x15, 0x20);  // MISC1: SRB1 activo
  writeRegister(0x17, 0x02);  // CONFIG4: comparador de lead-off

  delay(50);  // Estabilización
  sendCommand(RDATAC);
}

// -----------------------------------------------------------------------
// Lectura y envío de una muestra (8 canales)
// -----------------------------------------------------------------------
void readAndSendData() {
  digitalWrite(PIN_CS, LOW);

  // Status (3 bytes) - ignorados
  SPI.transfer(0x00);
  SPI.transfer(0x00);
  SPI.transfer(0x00);

  // 8 canales
  for (int i = 0; i < 8; i++) {
    byte b1 = SPI.transfer(0x00); // MSB
    byte b2 = SPI.transfer(0x00); // MID
    byte b3 = SPI.transfer(0x00); // LSB

    long val = (long)((b1 << 16) | (b2 << 8) | b3);
    if ((val & 0x800000) > 0) val |= 0xFF000000; // signo (complemento a 2)

    channelData[i] = val;
  }
  digitalWrite(PIN_CS, HIGH);

  memcpy(&eegPayload[2], channelData, 32);
  Serial.write(eegPayload, PACKET_SIZE);
}

// -----------------------------------------------------------------------
// Helpers SPI
// -----------------------------------------------------------------------
void sendCommand(byte cmd) {
  digitalWrite(PIN_CS, LOW);
  SPI.transfer(cmd);
  delayMicroseconds(5);
  digitalWrite(PIN_CS, HIGH);
}

void writeRegister(byte reg, byte val) {
  digitalWrite(PIN_CS, LOW);
  SPI.transfer(WREG | reg);
  SPI.transfer(0x00);
  SPI.transfer(val);
  delayMicroseconds(5);
  digitalWrite(PIN_CS, HIGH);
}

package dsp

// Biquad implementa una sección de segundo orden (Direct Form II Transpuesta)
type Biquad struct {
	b0, b1, b2 float64
	a1, a2     float64
	z1, z2     float64 // Memoria de retraso (estados)
}

// NewBiquad inicializa los coeficientes. (MATLAB siempre normaliza a0 a 1.0)
func NewBiquad(b0, b1, b2, a1, a2 float64) Biquad {
	return Biquad{b0: b0, b1: b1, b2: b2, a1: a1, a2: a2}
}

// ProcessSample procesa una sola muestra a través de este biquad
func (bq *Biquad) ProcessSample(x float64) float64 {
	y := (bq.b0 * x) + bq.z1
	bq.z1 = (bq.b1 * x) - (bq.a1 * y) + bq.z2
	bq.z2 = (bq.b2 * x) - (bq.a2 * y)
	return y
}

// IIRFilter contiene la cascada completa y la ganancia
type IIRFilter struct {
	sections []Biquad
	gain     float64
}

// NewIIRFilter crea el filtro a partir de la matriz SOS de MATLAB y la ganancia total
func NewIIRFilter(sos [][]float64, totalGain float64) *IIRFilter {
	filter := &IIRFilter{
		gain: totalGain,
	}
	for _, row := range sos {
		// MATLAB SOS format: [b0, b1, b2, a0, a1, a2]
		// Ignoramos a0 (row[3]) porque en la matriz SOS de MATLAB siempre es 1.0
		filter.sections = append(filter.sections, NewBiquad(row[0], row[1], row[2], row[4], row[5]))
	}
	return filter
}

// ProcessSample pasa la muestra por la ganancia y luego por toda la cascada
func (f *IIRFilter) ProcessSample(x float64) float64 {
	y := x * f.gain
	for i := range f.sections {
		y = f.sections[i].ProcessSample(y)
	}
	return y
}

// EMAFilter implementa un filtro exponencial de primer orden (IIR):
//   y[n] = alpha*x[n] + (1-alpha)*y[n-1]
// Equivale al filter(alpha, [1 -(1-alpha)], x) de MATLAB.
type EMAFilter struct {
	alpha float64
	beta  float64
	last  float64
}

func NewEMAFilter(alpha float64) *EMAFilter {
	return &EMAFilter{alpha: alpha, beta: 1.0 - alpha}
}

func (e *EMAFilter) Process(x float64) float64 {
	e.last = e.alpha*x + e.beta*e.last
	return e.last
}

func (e *EMAFilter) Reset() {
	e.last = 0
}

package dsp

import "sort"

// MedianFilter implementa un filtro mediana causal de orden N.
// Equivale a medfilt1(x, N) de MATLAB pero en forma causal (usa solo muestras pasadas).
// Operación: O(N log N) por muestra, despreciable a 250 Hz con N=50.
type MedianFilter struct {
	size   int
	buffer []float64
	index  int
	count  int
}

func NewMedianFilter(size int) *MedianFilter {
	return &MedianFilter{
		size:   size,
		buffer: make([]float64, size),
	}
}

func (m *MedianFilter) Process(x float64) float64 {
	m.buffer[m.index] = x
	m.index = (m.index + 1) % m.size
	if m.count < m.size {
		m.count++
	}

	// Copiar las muestras válidas y ordenar para encontrar la mediana
	tmp := make([]float64, m.count)
	for i := 0; i < m.count; i++ {
		idx := (m.index - 1 - i + m.size) % m.size
		tmp[i] = m.buffer[idx]
	}
	sort.Float64s(tmp)
	return tmp[m.count/2]
}

func (m *MedianFilter) Reset() {
	m.index = 0
	m.count = 0
	for i := range m.buffer {
		m.buffer[i] = 0
	}
}

package dsp

import (
	"encoding/csv"
	"fmt"
	"os"
	"strconv"
)

// FIRFilter mantiene el estado de un filtro para UN solo canal
type FIRFilter struct {
	coeffs []float64
	buffer []float64 // Buffer circular para guardar muestras pasadas
	index  int       // Puntero actual en el buffer
	order  int
}

// NewFIRFilter carga los coeficientes y prepara la memoria
func NewFIRFilter(coeffs []float64) *FIRFilter {
	order := len(coeffs)
	return &FIRFilter{
		coeffs: coeffs,
		buffer: make([]float64, order), // Inicializado en 0
		order:  order,
		index:  0,
	}
}

// ProcessSample aplica la convolución: y[n] = sum(h[k] * x[n-k])
func (f *FIRFilter) ProcessSample(input float64) float64 {
	// 1. Guardar nueva muestra en el buffer circular
	f.buffer[f.index] = input

	// 2. Convolución
	var output float64 = 0
	bufLen := f.order

	// Recorremos los coeficientes
	for i := 0; i < bufLen; i++ {
		// Calculamos el índice circular correspondiente a x[n-k]
		// La muestra más reciente está en f.index
		// La anterior en f.index-1, etc.
		idx := (f.index - i + bufLen) % bufLen
		output += f.coeffs[i] * f.buffer[idx]
	}

	// 3. Avanzar índice
	f.index = (f.index + 1) % bufLen

	return output
}

// LoadCoeffs lee el CSV de coeficientes
func LoadCoeffs(path string) ([]float64, error) {
	file, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer file.Close()

	reader := csv.NewReader(file)
	records, err := reader.ReadAll()
	if err != nil {
		return nil, err
	}

	var coeffs []float64
	for _, row := range records {
		if len(row) > 0 {
			val, err := strconv.ParseFloat(row[0], 64)
			if err == nil {
				coeffs = append(coeffs, val)
			}
		}
	}
	fmt.Printf("Filtro cargado: %d Taps (Coeficientes)\n", len(coeffs))
	return coeffs, nil
}

package dsp

// FiltFiltStatic aplica filtrado de fase cero (forward-backward) sobre un buffer
// estático, equivalente a filtfilt() de MATLAB. Duplica el orden efectivo del
// filtro y no introduce desfase. Usa padding por reflexión impar en los extremos
// para reducir transitorios de borde.
//
// Pensado para análisis offline sobre buffers ya capturados (p.ej. el test de
// latiguillos de 20 s), NO para procesamiento muestra a muestra en tiempo real.
func FiltFiltStatic(sos [][]float64, gain float64, x []float64) []float64 {
	n := len(x)
	out := make([]float64, n)
	if n == 0 || sos == nil {
		copy(out, x)
		return out
	}

	// Longitud de padding por reflexión (similar a MATLAB: ~3 * orden del filtro).
	padLen := 3 * (2*len(sos) + 1)
	if padLen >= n {
		padLen = n - 1
	}
	if padLen < 0 {
		padLen = 0
	}

	// Señal extendida con reflexión impar: xp = [2*x0 - x[pad..1], x, 2*xN - x[N-1..N-pad]]
	ext := make([]float64, 0, n+2*padLen)
	for i := padLen; i >= 1; i-- {
		ext = append(ext, 2*x[0]-x[i])
	}
	ext = append(ext, x...)
	for i := 1; i <= padLen; i++ {
		ext = append(ext, 2*x[n-1]-x[n-1-i])
	}

	// Pasada hacia adelante (estado inicial en cero, fresco).
	f1 := NewIIRFilter(sos, gain)
	for i := range ext {
		ext[i] = f1.ProcessSample(ext[i])
	}
	// Invertir, pasada hacia atrás, invertir de nuevo.
	reverseFloat64(ext)
	f2 := NewIIRFilter(sos, gain)
	for i := range ext {
		ext[i] = f2.ProcessSample(ext[i])
	}
	reverseFloat64(ext)

	// Quitar el padding de los extremos.
	copy(out, ext[padLen:padLen+n])
	return out
}

func reverseFloat64(a []float64) {
	for i, j := 0, len(a)-1; i < j; i, j = i+1, j-1 {
		a[i], a[j] = a[j], a[i]
	}
}

	med_C4_mu   *dsp.MedianFilter
	med_C3_beta *dsp.MedianFilter
	med_C4_beta *dsp.MedianFilter
	med_EMG     *dsp.MedianFilter

	// Filtros EMA (envolvente suave por banda)
	ema_C3_mu   *dsp.EMAFilter
	ema_C4_mu   *dsp.EMAFilter
	ema_C3_beta *dsp.EMAFilter
	ema_C4_beta *dsp.EMAFilter
	ema_EMG     *dsp.EMAFilter

	// Buffer histórico circular (n_base + n_gap + n_actual = 1750 muestras = 7s)
	hist_C3_pwr_mu   []float64
	hist_C4_pwr_mu   []float64
	hist_C3_pwr_beta []float64
	hist_C4_pwr_beta []float64
	hist_M_pwr       []float64

	// Línea base dinámica (equivalente a C3_reposo_prom_* de MATLAB)
	C3_reposo_prom_mu   float64
	C4_reposo_prom_mu   float64
	C3_reposo_prom_beta float64
	C4_reposo_prom_beta float64
	M_reposo_prom       float64
	M_reposo_std        float64

	// Umbrales EMG
	M_umbral_on  float64
	M_umbral_off float64

	// ERD actual (retenciones ZOH para emisión a Angular)
	erd_c3_actual_mu   float64
	erd_c4_actual_mu   float64
	erd_c3_actual_beta float64
	erd_c4_actual_beta float64

	// Correlaciones EMG-EEG (debug; ZOH entre cálculos cada step_n)
	corr_c3 float64
	corr_c4 float64

	// Caída de potencia mu respecto a la base (positivo = desincronización);
	// usado por la decisión winner-take-all entre C3 y C4
	drop_c3 float64
	drop_c4 float64

	// Comandos de motor
	comando_motor_c3 int
	comando_motor_c4 int

	// Último estado de motores enviado al ESP32 (bit0=C3, bit1=C4); -1 = sin enviar.
	// Evita reenviar el comando 'M' si el estado no cambió.
	ultimoEstadoMotores int

	// Persistencia (contador de ERD consecutivos)
	contador_c3 int
	contador_c4 int

	// Timeout de motor (ciclos encendido)
	timer_c3 int
	timer_c4 int

	// Estado EMG
	estado_mov_emg int

	// Detección de artefactos
	artefacto_c3 bool
	artefacto_c4 bool

	// Bloqueo de línea base (muestras restantes con base congelada)
	bloqueo_base_c3 int
	bloqueo_base_c4 int

	// Semáforo
	led_rojo_c3 bool
	led_rojo_c4 bool

	// Contador de pasos (actualiza cálculos cada step_n muestras)
	step_counter int

	// Guardar datos en memoria
	recordMutex  sync.Mutex
	recordedData [][]int32

	// Puerto serial compartido (para enviar comandos al ESP32: modo y motores)
	serialPort serial.Port
	serialMu   sync.Mutex

	// Resultado del último test de latiguillos
	leadoffResult models.LeadOffResult

	// Entrenamiento 10-10-10
	entrenamientoResult models.EntrenamientoResult
	entPend             pendienteEntrenamiento

	// Base fija entrenada (se usa en tiempo real en lugar de la base móvil)
	baselineEntrenado bool
	baseFijaC3Mu      float64
	baseFijaC3Beta    float64
	baseFijaC4Mu      float64
	baseFijaC4Beta    float64

	// simCalibLista indica que se cargó y procesó un CSV de calibración para la
	// simulación (entPend ya tiene bases/umbrales válidos listos para confirmar).
	simCalibLista bool
}

func (a *App) startup(ctx context.Context) {
	a.ctx = ctx
	a.params = config.DefaultParams()
	a.derived = config.Derive(a.params)
	a.inicializarFiltros()

	// Umbrales de slider iniciales (coinciden con defaults de MATLAB)
	a.slider_erd_mu = -20.0
	a.slider_erd_beta = -15.0
	a.slider_ers_beta = 20.0
	a.actualizarUmbralesDesdeSliders(a.slider_erd_mu, a.slider_erd_beta, a.slider_ers_beta)
}

// inicializarFiltros crea todos los filtros IIR, mediana y EMA a partir de params actuales.
func (a *App) inicializarFiltros() {
	d := a.derived
	p := a.params

	notchSOS, notchGain := dsp.GetFilterNotch60(p.Orden_notch)
	a.iir_notch60_C3 = dsp.NewIIRFilter(notchSOS, notchGain)
	a.iir_notch60_C4 = dsp.NewIIRFilter(notchSOS, notchGain)
	a.iir_notch60_EMG = dsp.NewIIRFilter(notchSOS, notchGain)

	// Highpass opcional: sólo se construye si el flag está activo.
	if p.Eliminar_ruido_mov_resp {
		hpSOS, hpGain := dsp.GetFilterHP(p.Orden_hp)
		a.iir_hp_C3 = dsp.NewIIRFilter(hpSOS, hpGain)
		a.iir_hp_C4 = dsp.NewIIRFilter(hpSOS, hpGain)
		a.iir_hp_EMG = dsp.NewIIRFilter(hpSOS, hpGain)
	} else {
		a.iir_hp_C3 = nil
		a.iir_hp_C4 = nil
		a.iir_hp_EMG = nil
	}

	muSOS, muGain := dsp.GetFilterMu(p.Orden_mu)
	a.iir_C3_mu = dsp.NewIIRFilter(muSOS, muGain)
	a.iir_C4_mu = dsp.NewIIRFilter(muSOS, muGain)

	betaSOS, betaGain := dsp.GetFilterBeta(p.Orden_beta)
	a.iir_C3_beta = dsp.NewIIRFilter(betaSOS, betaGain)
	a.iir_C4_beta = dsp.NewIIRFilter(betaSOS, betaGain)

	emgSOS, emgGain := dsp.GetFilterEMG(p.Orden_emg)
	a.iir_EMG = dsp.NewIIRFilter(emgSOS, emgGain)

	// Filtros mediana (n_median = 2*round(t_duracion_picos*fs))
	a.med_C3_mu = dsp.NewMedianFilter(d.N_median)
	a.med_C4_mu = dsp.NewMedianFilter(d.N_median)
	a.med_C3_beta = dsp.NewMedianFilter(d.N_median)
	a.med_C4_beta = dsp.NewMedianFilter(d.N_median)
	a.med_EMG = dsp.NewMedianFilter(d.N_median)

	// Filtros EMA (alpha configurable)
	a.ema_C3_mu = dsp.NewEMAFilter(p.Alpha)
	a.ema_C4_mu = dsp.NewEMAFilter(p.Alpha)
	a.ema_C3_beta = dsp.NewEMAFilter(p.Alpha)
	a.ema_C4_beta = dsp.NewEMAFilter(p.Alpha)
	a.ema_EMG = dsp.NewEMAFilter(p.Alpha)
}

// actualizarUmbralesDesdeSliders actualiza los umbrales logarítmicos desde los sliders de Angular.
// Los sliders envían porcentajes: ERD como negativos (-20 = 20% caída), ERS como positivo (+20).
func (a *App) actualizarUmbralesDesdeSliders(thErdMu, thErdBeta, thErsBeta float64) {
	a.paramsMu.Lock()
	defer a.paramsMu.Unlock()

	a.slider_erd_mu = thErdMu
	a.slider_erd_beta = thErdBeta
	a.slider_ers_beta = thErsBeta

	// ERD: el slider viene negativo (-20%) → log(1 + (-0.20)) = log(0.80) = -0.2231
	a.ERD_C3_mu_umbral = math.Log(1.0 + (thErdMu / 100.0))
	a.ERD_C4_mu_umbral = a.ERD_C3_mu_umbral
	a.ERD_C3_beta_umbral = math.Log(1.0 + (thErdBeta / 100.0))
	a.ERD_C4_beta_umbral = a.ERD_C3_beta_umbral

	// ERS: el slider viene positivo (+20%) → log(1 + 0.20) = 0.1823
	a.ERS_C3_beta_umbral = math.Log(1.0 + (thErsBeta / 100.0))
	a.ERS_C4_beta_umbral = a.ERS_C3_beta_umbral
}

func (a *App) IniciarStreaming(puerto string) string {
	if a.estadoActual != EstadoDetenido {
		return "Error: El sistema ya está en ejecución."
	}
	a.estadoActual = EstadoStreaming
	a.reiniciarEstado()

	// Abrir el puerto y poner el ESP32 en modo tiempo real (stream + motores),
	// confirmando que ya transmite antes de continuar.
	port, err := a.iniciarModo(puerto, 'R')
	if err != nil {
		a.estadoActual = EstadoDetenido
		return fmt.Sprintf("Error puerto: %v", err)
	}
	a.serialMu.Lock()
	a.serialPort = port
	a.serialMu.Unlock()

	// Si hay base entrenada, fijarla como referencia (en lugar de la base móvil).
	if a.baselineEntrenado {
		a.paramsMu.Lock()
		a.C3_reposo_prom_mu = a.baseFijaC3Mu
		a.C3_reposo_prom_beta = a.baseFijaC3Beta
		a.C4_reposo_prom_mu = a.baseFijaC4Mu
		a.C4_reposo_prom_beta = a.baseFijaC4Beta
		a.paramsMu.Unlock()
	}

	a.recordMutex.Lock()
	a.recordedData = make([][]int32, 0)
	a.recordMutex.Unlock()

	ctx, cancel := context.WithCancel(context.Background())
	a.cancelRead = cancel
	go a.leerDatos(ctx, port)

	return "Streaming Iniciado"
}

func (a *App) IniciarCalibracion() string {
	if a.estadoActual != EstadoStreaming {
		return "Error: Primero debe Iniciar Streaming."
	}
	a.recordMutex.Lock()
	a.recordedData = [][]int32{}
	a.recordMutex.Unlock()

	a.estadoActual = EstadoAdquiriendo
	a.step_counter = 0
	return "Adquisición Formal Iniciada (Guardando datos)..."
}

// ActualizarConfiguracion actualiza todos los parámetros desde el modal de Angular.
// Solo se puede llamar con el sistema detenido.
func (a *App) ActualizarConfiguracion(p models.BCIParams) string {
	if a.estadoActual != EstadoDetenido {
		return "Error: Detenga el streaming antes de cambiar la configuración."
	}
	// Validar que existan SOS para los órdenes solicitados
	if sos, _ := dsp.GetFilterMu(p.Orden_mu); sos == nil {
		return fmt.Sprintf("Error: No hay coeficientes SOS para filtro Mu orden %d.", p.Orden_mu)
	}
	if sos, _ := dsp.GetFilterBeta(p.Orden_beta); sos == nil {
		return fmt.Sprintf("Error: No hay coeficientes SOS para filtro Beta orden %d.", p.Orden_beta)
	}
	if sos, _ := dsp.GetFilterEMG(p.Orden_emg); sos == nil {
		return fmt.Sprintf("Error: No hay coeficientes SOS para filtro EMG orden %d.", p.Orden_emg)
	}
	if sos, _ := dsp.GetFilterNotch60(p.Orden_notch); sos == nil {
		return fmt.Sprintf("Error: No hay coeficientes SOS para Notch60 orden %d.", p.Orden_notch)
	}
	if p.Eliminar_ruido_mov_resp {
		if sos, _ := dsp.GetFilterHP(p.Orden_hp); sos == nil {
			return fmt.Sprintf("Error: No hay coeficientes SOS para Highpass orden %d.", p.Orden_hp)
		}
	}

	a.paramsMu.Lock()
	a.params = p
	a.derived = config.Derive(p)
	a.inicializarFiltros()
	a.paramsMu.Unlock()

	// Reiniciar historial, baselines y contadores para que los filtros nuevos
	// no arrastren estado de la configuración anterior.
	a.reiniciarEstado()
	return "Configuración actualizada correctamente."
}

func (a *App) Detener() string {
	if a.estadoActual == EstadoDetenido {
		return "Ya está detenido."
	}
	estadoAnterior := a.estadoActual
	a.estadoActual = EstadoDetenido

	// Volver el ESP32 a IDLE (prótesis a reposo 150°) antes de cerrar el puerto.
	a.enviarComando('I', 0)

	if a.cancelRead != nil {
		a.cancelRead()
	}

	// Cerrar el puerto de forma síncrona para liberarlo de inmediato (evita que
	// un "Probar latiguillos" inmediato falle con "port busy" si el goroutine
	// de lectura aún no ha corrido su cierre diferido).
	a.serialMu.Lock()
	if a.serialPort != nil {
		a.serialPort.Close()
		a.serialPort = nil
	}
	a.serialMu.Unlock()

	if estadoAnterior == EstadoCalibrando || estadoAnterior == EstadoAdquiriendo {
		return a.guardarArchivosDual()
	}
	return "Sesión cancelada."
}

// -----------------------------------------------------------------------
// Goroutine principal de lectura y procesamiento DSP
// -----------------------------------------------------------------------

func (a *App) leerDatos(ctx context.Context, port serial.Port) {
	// Cierre guardado: sólo cierra si el puerto sigue siendo éste (Detener puede
	// haberlo cerrado ya de forma síncrona). Evita doble cierre.
	defer a.cerrarPuerto(port)
	payload := make([]byte, 32)
	oneByte := make([]byte, 1)
	port.ResetInputBuffer()
	fmt.Println("--- INICIANDO LECTURA CON HEADER 0xA0 0xC0 ---")

	d := a.derived // Copia local de muestras derivadas

	for {
		// Fase 1: Búsqueda de sincronía (leerExacto revisa ctx.Done() y tolera
		// timeouts; devuelve error si el puerto se cierra o se cancela ctx).
		if err := a.leerExacto(ctx, port, oneByte, time.Time{}); err != nil {
			return
		}
		if oneByte[0] != 0xA0 {
			continue
		}
		if err := a.leerExacto(ctx, port, oneByte, time.Time{}); err != nil {
			return
		}
		if oneByte[0] != 0xC0 {
			continue
		}

		// Fase 2: Lectura de payload
		if err := a.leerExacto(ctx, port, payload, time.Time{}); err != nil {
			return
		}

		// Fase 3: Decodificar 8 canales crudos y procesar la muestra por la cadena DSP.
		filaRaw := make([]int32, 8)
		for i := 0; i < 8; i++ {
			filaRaw[i] = int32(binary.LittleEndian.Uint32(payload[i*4 : (i+1)*4]))
		}
		a.procesarMuestra(filaRaw, d)
	}
}

// procesarMuestra corre la cadena DSP completa sobre una muestra de 8 canales
// (crudos int32): mapea canales, aplica filtros, calcula potencias/ERD, ejecuta la
// máquina de estados de motores, envía el comando 'M' al ESP32 cuando cambia y emite
// la trama "bci-data" al frontend. La usan leerDatos (serial en vivo) y simularDatos
// (reproducción de un CSV), de modo que ambos comparten exactamente el mismo pipeline.
func (a *App) procesarMuestra(filaRaw []int32, d config.DerivedSamples) {
	{
		filaUV := make([]float64, 8)
		for i := 0; i < 8; i++ {
			filaUV[i] = float64(filaRaw[i]) * ScaleFactor
		}

		if a.estadoActual == EstadoCalibrando || a.estadoActual == EstadoAdquiriendo {
			a.recordMutex.Lock()
			a.recordedData = append(a.recordedData, filaRaw)
			a.recordMutex.Unlock()
		}

		// Mapeo de canales:
		// CH1:C3, CH2:C3', CH7:C3'', CH3:C4, CH4:C4', CH8:C4'', CH5:M1, CH6:M2
		c3 := filaUV[0]
		c3p := filaUV[1]
		c3pp := filaUV[6]
		c4 := filaUV[2]
		c4p := filaUV[3]
		c4pp := filaUV[7]
		m1 := filaUV[4]
		m2 := filaUV[5]

		// Laplaciano pequeño y diferencial EMG
		diffC3 := c3 - (c3p+c3pp)/2.0
		diffC4 := c4 - (c4p+c4pp)/2.0
		diffEMG := m1 - m2

		// ─── CADENA DSP (igual que MATLAB) ───────────────────────────
		// 0. Filtro Notch 60 Hz
		cleanC3 := a.iir_notch60_C3.ProcessSample(diffC3)
		cleanC4 := a.iir_notch60_C4.ProcessSample(diffC4)
		cleanEMG := a.iir_notch60_EMG.ProcessSample(diffEMG)

		// 0.5 Highpass respiratorio (opcional, antes del bandpass)
		if a.iir_hp_C3 != nil {
			cleanC3 = a.iir_hp_C3.ProcessSample(cleanC3)
			cleanC4 = a.iir_hp_C4.ProcessSample(cleanC4)
			cleanEMG = a.iir_hp_EMG.ProcessSample(cleanEMG)
		}

		// 1. Filtros Bandpass IIR
		filtC3Mu := a.iir_C3_mu.ProcessSample(cleanC3)
		filtC4Mu := a.iir_C4_mu.ProcessSample(cleanC4)
		filtC3Beta := a.iir_C3_beta.ProcessSample(cleanC3)
		filtC4Beta := a.iir_C4_beta.ProcessSample(cleanC4)
		filtEMG := a.iir_EMG.ProcessSample(cleanEMG)

		// 2. Potencia cruda (señal^2)
		// 3. Mediana (elimina picos de interferencia, equivale a medfilt1)
		// 4. EMA (envolvente suave, equivale a filter(alpha,[1-(1-alpha)],x))
		// 5. Logaritmo (escala log)
		pwr_c3_mu := math.Log(a.ema_C3_mu.Process(a.med_C3_mu.Process(filtC3Mu*filtC3Mu)) + 1e-8)
		pwr_c4_mu := math.Log(a.ema_C4_mu.Process(a.med_C4_mu.Process(filtC4Mu*filtC4Mu)) + 1e-8)
		pwr_c3_beta := math.Log(a.ema_C3_beta.Process(a.med_C3_beta.Process(filtC3Beta*filtC3Beta)) + 1e-8)
		pwr_c4_beta := math.Log(a.ema_C4_beta.Process(a.med_C4_beta.Process(filtC4Beta*filtC4Beta)) + 1e-8)
		pwr_M := math.Log(a.ema_EMG.Process(a.med_EMG.Process(filtEMG*filtEMG)) + 1e-8)

		// ─── BUFFER HISTÓRICO (n_base + n_gap + n_actual = 1750 muestras) ───
		a.hist_C3_pwr_mu = append(a.hist_C3_pwr_mu, pwr_c3_mu)
		a.hist_C4_pwr_mu = append(a.hist_C4_pwr_mu, pwr_c4_mu)
		a.hist_C3_pwr_beta = append(a.hist_C3_pwr_beta, pwr_c3_beta)
		a.hist_C4_pwr_beta = append(a.hist_C4_pwr_beta, pwr_c4_beta)
		a.hist_M_pwr = append(a.hist_M_pwr, pwr_M)

		if len(a.hist_C3_pwr_mu) > d.Hist_size {
			a.hist_C3_pwr_mu = a.hist_C3_pwr_mu[1:]
			a.hist_C4_pwr_mu = a.hist_C4_pwr_mu[1:]
			a.hist_C3_pwr_beta = a.hist_C3_pwr_beta[1:]
			a.hist_C4_pwr_beta = a.hist_C4_pwr_beta[1:]
			a.hist_M_pwr = a.hist_M_pwr[1:]
		}

		// ─── MÁQUINA DE ESTADOS (solo cuando el historial está lleno) ───
		switch a.estadoActual {
		case EstadoStreaming, EstadoAdquiriendo, EstadoSimulando:
			a.step_counter++
			if a.step_counter >= d.Step_n {
				a.step_counter = 0

				if len(a.hist_C3_pwr_mu) == d.Hist_size {
					// Snapshot bajo lock: umbrales y params pueden cambiar
					// concurrentemente desde ActualizarUmbrales/ActualizarConfiguracion.
					a.paramsMu.RLock()
					umbCorr := a.params.Umbral_correlacion
					umbRuidoExt := a.derived.Umbral_ruido_extremo
					kPersist := a.params.K_persistencia
					kEmgOn := a.params.K_emg_on
					kEmgOff := a.params.K_emg_off
					thErdC3Mu := a.ERD_C3_mu_umbral
					thErdC4Mu := a.ERD_C4_mu_umbral
					thErdC3Beta := a.ERD_C3_beta_umbral
					thErdC4Beta := a.ERD_C4_beta_umbral
					thErsC3Beta := a.ERS_C3_beta_umbral
					thErsC4Beta := a.ERS_C4_beta_umbral
					baselineFijo := a.baselineEntrenado
					a.paramsMu.RUnlock()

					// Ventana actual = últimas n_actual muestras del historial
					rango_actual_C3 := a.hist_C3_pwr_mu[d.Hist_size-d.N_actual:]
					rango_actual_C4 := a.hist_C4_pwr_mu[d.Hist_size-d.N_actual:]
					rango_actual_EMG := a.hist_M_pwr[d.Hist_size-d.N_actual:]
					rango_actual_C3_beta := a.hist_C3_pwr_beta[d.Hist_size-d.N_actual:]
					rango_actual_C4_beta := a.hist_C4_pwr_beta[d.Hist_size-d.N_actual:]

					// Ventana de análisis (promedio de n_actual muestras = prom_mov_c3_mu en MATLAB)
					prom_mov_c3_mu := calcularMedia(rango_actual_C3)
					prom_mov_c4_mu := calcularMedia(rango_actual_C4)
					prom_mov_c3_beta := calcularMedia(rango_actual_C3_beta)
					prom_mov_c4_beta := calcularMedia(rango_actual_C4_beta)

					// Ventana base = n_base muestras separadas por n_gap de la actual
					fin_base := d.Hist_size - d.N_actual - d.N_gap
					rango_base_C3_mu := a.hist_C3_pwr_mu[fin_base-d.N_base : fin_base]
					rango_base_C4_mu := a.hist_C4_pwr_mu[fin_base-d.N_base : fin_base]
					rango_base_C3_beta := a.hist_C3_pwr_beta[fin_base-d.N_base : fin_base]
					rango_base_C4_beta := a.hist_C4_pwr_beta[fin_base-d.N_base : fin_base]

					// ── 6.1 DETECCIÓN DE ARTEFACTOS POR CORRELACIÓN ──
					win_m_actual := rango_actual_EMG
					prom_mov_m := calcularMedia(win_m_actual)

					// Estado EMG (histeresis)
					if prom_mov_m > a.M_umbral_on {
						a.estado_mov_emg = 1
					} else if prom_mov_m < a.M_umbral_off {
						a.estado_mov_emg = 0
					}

					// Decrementar bloqueos
					a.bloqueo_base_c3 = max(0, a.bloqueo_base_c3-d.Step_n)
					a.bloqueo_base_c4 = max(0, a.bloqueo_base_c4-d.Step_n)

					a.corr_c3 = calcularPearson(rango_actual_C3, win_m_actual)
					a.corr_c4 = calcularPearson(rango_actual_C4, win_m_actual)

					a.artefacto_c3 = math.Abs(a.corr_c3) > umbCorr
					a.artefacto_c4 = math.Abs(a.corr_c4) > umbCorr

					if a.artefacto_c3 {
						a.bloqueo_base_c3 = d.Muestras_bloqueo
					}
					if a.artefacto_c4 {
						a.bloqueo_base_c4 = d.Muestras_bloqueo
					}

					// ── 6.2 ERD ACTUAL ──
					a.erd_c3_actual_mu = prom_mov_c3_mu - a.C3_reposo_prom_mu
					a.erd_c4_actual_mu = prom_mov_c4_mu - a.C4_reposo_prom_mu
					a.erd_c3_actual_beta = prom_mov_c3_beta - a.C3_reposo_prom_beta
					a.erd_c4_actual_beta = prom_mov_c4_beta - a.C4_reposo_prom_beta

					// ── 6.3 DETECCIÓN DE ARTEFACTOS POR PICOS GIGANTES ──
					if a.erd_c3_actual_mu > umbRuidoExt && a.erd_c3_actual_beta > umbRuidoExt {
						a.artefacto_c3 = true
						a.bloqueo_base_c3 = d.Muestras_bloqueo
					}
					if a.erd_c4_actual_mu > umbRuidoExt && a.erd_c4_actual_beta > umbRuidoExt {
						a.artefacto_c4 = true
						a.bloqueo_base_c4 = d.Muestras_bloqueo
					}

					// ── SEMÁFORO ──
					a.led_rojo_c3 = a.artefacto_c3 || a.bloqueo_base_c3 > 0
					a.led_rojo_c4 = a.artefacto_c4 || a.bloqueo_base_c4 > 0

					// ── 6.4 LÍNEA BASE DINÁMICA ──
					// Si baselineFijo (base entrenada): NO se actualiza (se fijó al iniciar streaming).
					// Si no: base móvil clásica, actualiza solo si motor apagado y sin bloqueo.
					if !baselineFijo && a.comando_motor_c3 == 0 && a.bloqueo_base_c3 == 0 {
						a.C3_reposo_prom_mu = calcularMedia(rango_base_C3_mu)
						a.C3_reposo_prom_beta = calcularMedia(rango_base_C3_beta)
					}
					if !baselineFijo && a.comando_motor_c4 == 0 && a.bloqueo_base_c4 == 0 {
						a.C4_reposo_prom_mu = calcularMedia(rango_base_C4_mu)
						a.C4_reposo_prom_beta = calcularMedia(rango_base_C4_beta)
					}

					// Primera calibración automática (después de llenar el historial):
					// calcular umbrales EMG con media y desviación estándar
					if a.M_umbral_on == 0 {
						rango_base_M := a.hist_M_pwr[fin_base-d.N_base : fin_base]
						a.M_reposo_prom, a.M_reposo_std = calcularMediaStd(rango_base_M)
						a.M_umbral_on = a.M_reposo_prom + kEmgOn*a.M_reposo_std
						a.M_umbral_off = a.M_reposo_prom + kEmgOff*a.M_reposo_std
					}

					// ── 6.5 MÁQUINA DE ESTADOS COMBINADA (EXCLUYENTE / WINNER-TAKE-ALL) ──
					// La misma lógica de timeout/ERS/artefacto y persistencia, pero ahora
					// SOLO un motor puede estar activo a la vez: cuando ambos hemisferios
					// cumplen la condición de ERD, gana el que MÁS cayó (mayor drop en mu).

					// Caída de potencia mu respecto a la base (positivo = desincronización)
					a.drop_c3 = a.C3_reposo_prom_mu - prom_mov_c3_mu
					a.drop_c4 = a.C4_reposo_prom_mu - prom_mov_c4_mu

					// Apagado por hemisferio (timeout, rebote ERS o artefacto)
					if a.comando_motor_c3 == 1 {
						a.timer_c3++
						if a.timer_c3 >= d.Max_ciclos_on || a.erd_c3_actual_beta > thErsC3Beta || a.artefacto_c3 {
							a.comando_motor_c3 = 0
							a.contador_c3 = 0
							a.timer_c3 = 0
						}
					}
					if a.comando_motor_c4 == 1 {
						a.timer_c4++
						if a.timer_c4 >= d.Max_ciclos_on || a.erd_c4_actual_beta > thErsC4Beta || a.artefacto_c4 {
							a.comando_motor_c4 = 0
							a.contador_c4 = 0
							a.timer_c4 = 0
						}
					}

					// Encendido SOLO si ningún motor está activo (exclusión mutua)
					if a.comando_motor_c3 == 0 && a.comando_motor_c4 == 0 {
						intentC3 := a.erd_c3_actual_mu < thErdC3Mu && a.erd_c3_actual_beta < thErdC3Beta && !a.artefacto_c3
						intentC4 := a.erd_c4_actual_mu < thErdC4Mu && a.erd_c4_actual_beta < thErdC4Beta && !a.artefacto_c4

						// Persistencia por hemisferio (se resetea ante rebote ERS o artefacto)
						if intentC3 {
							a.contador_c3++
						} else if a.erd_c3_actual_beta > thErsC3Beta || a.artefacto_c3 {
							a.contador_c3 = 0
						}
						if intentC4 {
							a.contador_c4++
						} else if a.erd_c4_actual_beta > thErsC4Beta || a.artefacto_c4 {
							a.contador_c4 = 0
						}

						c3Listo := intentC3 && a.contador_c3 >= kPersist
						c4Listo := intentC4 && a.contador_c4 >= kPersist

						switch {
						case c3Listo && c4Listo:
							// Ambos cumplen: gana el hemisferio que más cayó (mayor drop)
							if a.drop_c3 >= a.drop_c4 {
								a.comando_motor_c3 = 1
								a.timer_c3 = 0
								a.contador_c4 = 0
							} else {
								a.comando_motor_c4 = 1
								a.timer_c4 = 0
								a.contador_c3 = 0
							}
						case c3Listo:
							a.comando_motor_c3 = 1
							a.timer_c3 = 0
							a.contador_c4 = 0
						case c4Listo:
							a.comando_motor_c4 = 1
							a.timer_c4 = 0
							a.contador_c3 = 0
						}
					}
				}
			}
		}

		// Snapshot bajo lock: umbrales pueden cambiar concurrentemente
		// desde ActualizarUmbrales/ActualizarConfiguracion.
		a.paramsMu.RLock()
		thEmg := a.M_umbral_on
		thErdC3 := a.ERD_C3_mu_umbral
		thErdC4 := a.ERD_C4_mu_umbral
		thErdC3B := a.ERD_C3_beta_umbral
		thErsC3B := a.ERS_C3_beta_umbral
		thErdC4B := a.ERD_C4_beta_umbral
		thErsC4B := a.ERS_C4_beta_umbral
		baselineFijo := a.baselineEntrenado
		a.paramsMu.RUnlock()

		// Emitir trama a Angular
		frame := models.BCIFrame{
			Estado: a.estadoActual,
			EMG:    pwr_M,
			ThEMG:  thEmg,

			ThErdC3:  thErdC3,
			ThErdC4:  thErdC4,
			ThErdC3B: thErdC3B,
			ThErsC3B: thErsC3B,
			ThErdC4B: thErdC4B,
			ThErsC4B: thErsC4B,

			BaselineFijo: baselineFijo,
		}

		if len(a.hist_C3_pwr_mu) < d.Hist_size {
			// Historial aún llenándose: ceros en ERD
			frame.ErdC3, frame.ErdC4 = 0, 0
			frame.ErdC3B, frame.ErdC4B = 0, 0
			frame.MotorC3, frame.MotorC4 = 0, 0
			frame.LedRojoC3, frame.LedRojoC4 = true, true
		} else {
			frame.ErdC3 = a.erd_c3_actual_mu
			frame.ErdC4 = a.erd_c4_actual_mu
			frame.ErdC3B = a.erd_c3_actual_beta
			frame.ErdC4B = a.erd_c4_actual_beta
			frame.MotorC3 = a.comando_motor_c3
			frame.MotorC4 = a.comando_motor_c4
			frame.LedRojoC3 = a.led_rojo_c3
			frame.LedRojoC4 = a.led_rojo_c4
			frame.ArtifactoC3 = a.artefacto_c3
			frame.ArtifactoC4 = a.artefacto_c4
			frame.CorrC3 = a.corr_c3
			frame.CorrC4 = a.corr_c4
			frame.DropC3 = a.drop_c3
			frame.DropC4 = a.drop_c4
		}

		// Enviar comando de motores al ESP32 SOLO cuando cambia el estado
		// (bit0=C3, bit1=C4). El firmware lo aplica únicamente en modo REALTIME.
		estadoMotores := a.comando_motor_c3 | (a.comando_motor_c4 << 1)
		if estadoMotores != a.ultimoEstadoMotores {
			a.ultimoEstadoMotores = estadoMotores
			a.enviarComando('M', byte(estadoMotores))
		}

		runtime.EventsEmit(a.ctx, "bci-data", frame)
	}
}

// errSinDatos indica que el ESP32 no envió ninguna trama dentro del plazo
// esperado (cable desconectado, firmware sin cargar o puerto equivocado).
var errSinDatos = errors.New("el ESP32 no envió datos")

// Buffer de diagnóstico del test de latiguillos. Se muestra en pantalla (evento
// "leadoff-debug") para que el usuario pueda copiarlo, y se respalda en un .txt
// junto al .exe. Es global para no tocar la struct App.
var (
	diagMu  sync.Mutex
	diagBuf []string
)

// reiniciarDiag limpia el buffer de diagnóstico (al empezar una prueba nueva).
func reiniciarDiag() {
	diagMu.Lock()
	diagBuf = nil
	diagMu.Unlock()
}

// debugLog agrega una línea con marca de tiempo al buffer de diagnóstico, la
// emite a la interfaz (cuadro copiable del modal) y la respalda en
// "latiguillos-debug.txt" junto al ejecutable. El build de producción no tiene
// consola, así que ésta es la forma de ver qué pasa durante la prueba.
func (a *App) debugLog(format string, args ...interface{}) {
	line := time.Now().Format("15:04:05.000") + "  " + fmt.Sprintf(format, args...)

	diagMu.Lock()
	diagBuf = append(diagBuf, line)
	txt := strings.Join(diagBuf, "\n")
	diagMu.Unlock()

	// Respaldo en archivo junto al .exe (por si prefieren enviar el .txt).
	if exe, err := os.Executable(); err == nil {
		ruta := filepath.Join(filepath.Dir(exe), "latiguillos-debug.txt")
		if f, ferr := os.OpenFile(ruta, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644); ferr == nil {
			fmt.Fprintln(f, line)
			f.Close()
		}
	}

	// Mostrar en pantalla (cuadro copiable del modal de latiguillos).
	if a.ctx != nil {
		runtime.EventsEmit(a.ctx, "leadoff-debug", txt)
	}
}

// abrirPuerto abre el puerto serial, fuerza un reinicio por hardware del ESP32
// y lo deja listo para leer. El reinicio (vía DTR/RTS, el circuito auto-reset de
// las placas ESP32) garantiza que la placa arranque limpia y acepte el comando
// de modo SIN IMPORTAR en qué estado quedó la sesión anterior (p. ej. en IDLE
// tras un 'I'), que era la causa de que a veces no respondiera (0 bytes).
// También fija un timeout de lectura para que los lazos puedan revisar
// ctx.Done() (sin él Read se bloquea indefinidamente → puerto que no se libera).
func (a *App) abrirPuerto(puerto string) (serial.Port, error) {
	mode := &serial.Mode{BaudRate: 921600}
	port, err := serial.Open(puerto, mode)
	if err != nil {
		return nil, err
	}
	// Con timeout, Read devuelve (0, nil) al expirar sin datos (ver leerExacto).
	if err := port.SetReadTimeout(200 * time.Millisecond); err != nil {
		port.Close()
		return nil, err
	}

	// Reinicio por hardware: pulso en EN (RTS) manteniendo GPIO0 alto (DTR=false)
	// para que arranque el firmware normal (no el bootloader). Si la placa no
	// tiene el circuito auto-reset, estas líneas no hacen daño y el reintento de
	// comando del handshake sigue cubriendo el arranque.
	port.SetDTR(false) // GPIO0 alto = arranque normal
	port.SetRTS(true)  // EN bajo  = placa en reset
	time.Sleep(120 * time.Millisecond)
	port.SetRTS(false) // EN alto  = la placa arranca
	a.debugLog("abrirPuerto(%s): reset por hardware enviado, esperando arranque del ESP32", puerto)

	// Esperar a que el ESP32 termine setup() (SPI delay 500ms + init).
	time.Sleep(1500 * time.Millisecond)
	port.ResetInputBuffer()
	return port, nil
}

// iniciarModo abre el puerto, pone al ESP32 en el modo pedido y confirma que
// está transmitiendo antes de devolver el puerto. Devuelve un error claro si el
// ESP32 no responde (en vez de dejar la app congelada).
func (a *App) iniciarModo(puerto string, cmd byte) (serial.Port, error) {
	port, err := a.abrirPuerto(puerto)
	if err != nil {
		return nil, err
	}
	if err := a.handshakeModo(port, cmd); err != nil {
		port.Close()
		return nil, err
	}
	return port, nil
}

// handshakeModo envía el comando de modo [cmd][0] y espera la primera cabecera
// 0xA0 0xC0. Reenvía el comando cada 500 ms hasta 4 s: el ESP32 unificado arranca
// en IDLE y puede perder el primer comando si éste llega mientras aún bootea, así
// que insistimos hasta ver datos reales.
func (a *App) handshakeModo(port serial.Port, cmd byte) error {
	inicio := time.Now()
	deadline := inicio.Add(4 * time.Second)
	var ultimoEnvio time.Time
	var prev byte
	bytesRecibidos := 0
	envios := 0
	buf := make([]byte, 1)
	a.debugLog("handshake('%c'): inicio", cmd)
	for time.Now().Before(deadline) {
		if time.Since(ultimoEnvio) >= 500*time.Millisecond {
			port.Write([]byte{cmd, 0})
			ultimoEnvio = time.Now()
			envios++
		}
		n, err := port.Read(buf)
		if err != nil {
			a.debugLog("handshake('%c'): error de lectura tras %d bytes: %v", cmd, bytesRecibidos, err)
			return err
		}
		if n == 0 {
			continue // timeout de lectura: reintentar
		}
		bytesRecibidos++
		if prev == 0xA0 && buf[0] == 0xC0 {
			a.debugLog("handshake('%c'): OK en %v (%d bytes, %d envios)", cmd,
				time.Since(inicio).Round(time.Millisecond), bytesRecibidos, envios)
			return nil // el ESP32 ya transmite en el modo pedido
		}
		prev = buf[0]
	}
	a.debugLog("handshake('%c'): TIMEOUT 4s — %d bytes recibidos, %d envios, sin cabecera 0xA0 0xC0",
		cmd, bytesRecibidos, envios)
	return fmt.Errorf("%w (0 datos). Soluciones: 1) presiona el botón EN/RST del ESP32 (o desconéctalo y vuelve a conectar el USB) e inténtalo de nuevo; 2) cierra el Monitor Serie de Arduino si está abierto; 3) verifica que el puerto sea el correcto y que el firmware ESP32_BCI esté cargado", errSinDatos)
}

// leerExacto llena buf leyendo del puerto. Tolera los timeouts de lectura
// (Read devuelve 0, nil) y revisa ctx.Done() entre intentos, de modo que la
// cancelación funciona aunque se esté a media trama. Si deadline no es cero y se
// supera sin completar la lectura, devuelve errSinDatos.
func (a *App) leerExacto(ctx context.Context, port serial.Port, buf []byte, deadline time.Time) error {
	n := 0
	for n < len(buf) {
		select {
		case <-ctx.Done():
			return ctx.Err()
		default:
		}
		if !deadline.IsZero() && time.Now().After(deadline) {
			return errSinDatos
		}
		m, err := port.Read(buf[n:])
		if err != nil {
			return err
		}
		n += m
	}
	return nil
}

// cerrarPuerto cierra el puerto y lo desreferencia sólo si sigue siendo el mismo
// que se pasó (evita cerrar un puerto que ya abrió una sesión más nueva). Es
// seguro llamarla varias veces.
func (a *App) cerrarPuerto(port serial.Port) {
	a.serialMu.Lock()
	defer a.serialMu.Unlock()
	if a.serialPort == port {
		a.serialPort.Close()
		a.serialPort = nil
	}
}

// enviarComando manda un mensaje de 2 bytes [cmd][arg] al ESP32 (modo o motores).
func (a *App) enviarComando(cmd, arg byte) {
	a.serialMu.Lock()
	defer a.serialMu.Unlock()
	if a.serialPort != nil {
		a.serialPort.Write([]byte{cmd, arg})
	}
}

// ProbarLatiguillos abre el puerto, pone el ESP32 en modo lead-off y captura 20 s.
func (a *App) ProbarLatiguillos(puerto string) string {
	if a.estadoActual != EstadoDetenido {
		return "Error: detenga el sistema antes de probar latiguillos."

		return "Error: faltan los coeficientes SOS lead-off (29-33 Hz). Pégalos en dsp/sos_tables.go (SOS_leadoff[4] y Gain_leadoff[4])."
	}
	// Abrir el puerto y poner el ESP32 en modo lead-off, confirmando que ya
	// transmite antes de iniciar la captura.
	reiniciarDiag()
	a.debugLog("ProbarLatiguillos: abriendo puerto %s en modo 'L'", puerto)
	port, err := a.iniciarModo(puerto, 'L')
	if err != nil {

	ctx, cancel := context.WithCancel(context.Background())
	a.cancelRead = cancel
	go a.capturarLeadoff(ctx, port)
	return "Probando latiguillos (20 s)..."
}

// RepetirLatiguillos relanza la captura de 20 s (puerto ya abierto en modo lead-off).

	a.cancelRead = cancel
	go a.capturarLeadoff(ctx, port)
	return "Repitiendo prueba de latiguillos (20 s)..."
}

// capturarLeadoff lee 20 s de tramas crudas y calcula la tabla de impedancias.
func (a *App) capturarLeadoff(ctx context.Context, port serial.Port) {
	const nObjetivo = 20 * config.Fs // 5000 muestras

	// Dejar estabilizar el lead-off del ADS1299 tras enviar 'L'.
	select {
	case <-ctx.Done():
		return
	case <-time.After(800 * time.Millisecond):
	}
	port.ResetInputBuffer()

	a.recordMutex.Lock()
	a.recordedData = make([][]int32, 0, nObjetivo)
	a.recordMutex.Unlock()

	payload := make([]byte, 32)
	oneByte := make([]byte, 1)
	capturadas := 0
	bytesRecibidos := 0
	headersA0 := 0
	ultimaTrama := time.Now()
	a.debugLog("capturarLeadoff: inicio (objetivo %d muestras)", nObjetivo)

	for capturadas < nObjetivo {
		// El plazo NO se reinicia en cada byte: si pasan 3 s sin completar una
		// trama válida (sin datos, o datos sin cabecera 0xA0 0xC0), abortamos con
		// un error visible en lugar de girar en silencio consumiendo bytes.
		dl := ultimaTrama.Add(3 * time.Second)
		if err := a.leerExacto(ctx, port, oneByte, dl); err != nil {
			a.abortarLeadoff(ctx, port, err, bytesRecibidos, headersA0, capturadas)
			return
		}
		bytesRecibidos++
		if oneByte[0] != 0xA0 {
			continue
		}
		headersA0++
		if err := a.leerExacto(ctx, port, oneByte, dl); err != nil {
			a.abortarLeadoff(ctx, port, err, bytesRecibidos, headersA0, capturadas)
			return
		}
		if oneByte[0] != 0xC0 {
			continue
		}
		if err := a.leerExacto(ctx, port, payload, dl); err != nil {
			a.abortarLeadoff(ctx, port, err, bytesRecibidos, headersA0, capturadas)
			return
		}
		fila := make([]int32, 8)
		for i := 0; i < 8; i++ {
			fila[i] = int32(binary.LittleEndian.Uint32(payload[i*4 : (i+1)*4]))
		}
		a.recordMutex.Lock()
		a.recordedData = append(a.recordedData, fila)
		a.recordMutex.Unlock()

		if capturadas == 0 {
			a.debugLog("capturarLeadoff: primera trama OK tras %d bytes / %d cabeceras 0xA0; payload: % X",
				bytesRecibidos, headersA0, payload)
		}
		capturadas++
		ultimaTrama = time.Now()
		if capturadas%config.Fs == 0 {
			seg := capturadas / config.Fs
			a.debugLog("capturarLeadoff: %d s (%d muestras)", seg, capturadas)
			runtime.EventsEmit(a.ctx, "leadoff-progress", seg)
		}
	}
	a.debugLog("capturarLeadoff: captura completa (%d muestras)", capturadas)

	a.recordMutex.Lock()
	datos := make([][]int32, len(a.recordedData))
	copy(datos, a.recordedData)
	a.recordMutex.Unlock()

	resultado := a.calcularLeadoff(datos)

	a.debugLog("calcularLeadoff: %d canales calculados", len(resultado.Filas))
	for _, f := range resultado.Filas {
		a.debugLog("  %-10s Vpp=%8.2f uV   Z=%8.2f kOhm   %s", f.Canal, f.VppUv, f.ZkOhm, f.Calidad)
	}

	a.recordMutex.Lock()
	a.leadoffResult = resultado
	a.recordMutex.Unlock()

	runtime.EventsEmit(a.ctx, "leadoff-result", resultado)
	a.debugLog("capturarLeadoff: tabla emitida a la interfaz (leadoff-result). FIN OK")
}

// abortarLeadoff maneja un fallo de lectura durante la captura de latiguillos.
// Si la causa fue la cancelación del contexto (Repetir/Confirmar), no hace nada:
// la nueva sesión o el cierre se encargan. Ante cualquier otro error (sin datos
// o puerto cerrado) libera el puerto, vuelve a estado detenido y avisa al
// frontend con "leadoff-error" para que el modal salga del estado "Probando...".
func (a *App) abortarLeadoff(ctx context.Context, port serial.Port, err error, bytesRecibidos, headersA0, capturadas int) {
	if ctx.Err() != nil {
		a.debugLog("capturarLeadoff: cancelado por ctx (Repetir/Confirmar/Detener)")
		return // cancelación intencional, no es un fallo
	}
	a.debugLog("capturarLeadoff: ABORTA err=%v bytes=%d headersA0=%d capturadas=%d",
		err, bytesRecibidos, headersA0, capturadas)
	a.cerrarPuerto(port)
	a.estadoActual = EstadoDetenido

	var msg string
	switch {
	case bytesRecibidos == 0:
		msg = "El ESP32 no envía datos en modo lead-off (0 bytes recibidos). Revisa la conexión y reinicia el ESP32."
	case headersA0 == 0:
		msg = fmt.Sprintf("Llegan datos del ESP32 (%d bytes) pero sin la cabecera 0xA0 esperada. Reinicia el ESP32 e inténtalo de nuevo.", bytesRecibidos)
	case capturadas == 0:
		msg = fmt.Sprintf("Llegan cabeceras (%d × 0xA0) pero ninguna trama se completa: desincronía en modo lead-off. Reinicia el ESP32 e inténtalo de nuevo.", headersA0)
	default:
		msg = "El ESP32 dejó de enviar datos a mitad de la prueba. Revisa la conexión USB e inténtalo de nuevo."
	}
	runtime.EventsEmit(a.ctx, "leadoff-error", msg)
}

// calcularLeadoff replica leadsoff.m: por canal, filtra 29-33 Hz (filtfilt),
// mide V_rms = std, V_pp = V_rms*2*sqrt(2) y Z = V_pp/(2*I)/1000 en kΩ.
func (a *App) calcularLeadoff(datos [][]int32) models.LeadOffResult {
	gv := (VRefLeadoff / (Gain * Resolution)) * 1e6 // µV por cuenta (VRef=4.5)
	sos, gain := dsp.GetFilterLeadoff(OrdenLeadoff)

	labels := []string{
		"CH1 (C3)", "CH2 (C3')", "CH3 (C4)", "CH4 (C4')",
		"CH5 (M1)", "CH6 (M2)", "CH7 (C3'')", "CH8 (C4'')",
	}

	n := len(datos)
	filas := make([]models.LeadOffRow, 0, 8)
	for ch := 0; ch < 8; ch++ {
		x := make([]float64, n)
		for i := 0; i < n; i++ {
			x[i] = float64(datos[i][ch]) * gv
		}
		// detrend: restar la media
		m := calcularMedia(x)
		for i := range x {
			x[i] -= m
		}
		xf := dsp.FiltFiltStatic(sos, gain, x)
		_, vrms := calcularMediaStd(xf)
		vpp := vrms * 2 * math.Sqrt2
		zKOhm := (vpp * 1e-6) / (2 * ILeadoff) / 1000

		filas = append(filas, models.LeadOffRow{
			Canal:   labels[ch],
			VppUv:   vpp,
			ZkOhm:   zKOhm,
			Calidad: calidadLeadoff(zKOhm),
		})
	}
	return models.LeadOffResult{Filas: filas}
}

// calidadLeadoff clasifica la impedancia con los mismos umbrales de leadsoff.m.
func calidadLeadoff(zKOhm float64) string {
	switch {
	case zKOhm < 5:
		return "✓ Excelente"
	case zKOhm < 10:
		return "✓ Buena"
	case zKOhm < 25:
		return "~ Marginal"
	case zKOhm < 100:
		return "✗ Mala"
	default:
		return "✗✗ Desconectado/muy mala"
	}
}

// escribirCSVcanales escribe los datos crudos int32 con el encabezado de canal
// estándar (compatible con leadsoff.m y protocolo.m).
func escribirCSVcanales(ruta string, datos [][]int32) error {
	f, err := os.Create(ruta)
	if err != nil {
		return err
	}
	defer f.Close()
	w := csv.NewWriter(f)
	w.Write([]string{"CH1_C3", "CH2_C3p", "CH3_C4", "CH4_C4p", "CH5_M1", "CH6_M2", "CH7_C3pp", "CH8_C4pp"})
	for _, fila := range datos {
		s := make([]string, len(fila))
		for i, v := range fila {
			s[i] = strconv.FormatInt(int64(v), 10)
		}
		w.Write(s)
	}
	w.Flush()
	return w.Error()
}

// leerCSVcrudo lee un CSV de 8 canales en crudo (int32 por columna, el formato que
// genera escribirCSVcanales/guardarArchivosDual) y devuelve las muestras como
// [][]int32. Acepta las filas que parsean a 8 enteros y descarta el resto, de modo
// que la cabecera (y un eventual BOM UTF-8 en ella) se omite automáticamente.
func leerCSVcrudo(ruta string) ([][]int32, error) {
	f, err := os.Open(ruta)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	r := csv.NewReader(f)
	r.FieldsPerRecord = -1 // validamos el ancho de cada fila a mano
	filas, err := r.ReadAll()
	if err != nil {
		return nil, err
	}

	datos := make([][]int32, 0, len(filas))
	for _, fila := range filas {
		if len(fila) < 8 {
			continue
		}
		muestra := make([]int32, 8)
		ok := true
		for c := 0; c < 8; c++ {
			v, perr := strconv.ParseInt(strings.TrimSpace(fila[c]), 10, 32)
			if perr != nil {
				ok = false
				break
			}
			muestra[c] = int32(v)
		}
		if ok {
			datos = append(datos, muestra)
		}
	}
	if len(datos) == 0 {
		return nil, fmt.Errorf("no se pudo leer ninguna muestra de 8 canales int32 (revisa que el CSV sea del formato crudo CH1_C3..CH8_C4pp)")
	}
	return datos, nil
}

// CargarCalibracionSim lee un CSV de calibración 10-10-10 y calcula bases y umbrales
// con el MISMO procesamiento que el entrenamiento en vivo (procesarEntrenamiento),
// emitiendo "entrenamiento-result" para reutilizar el modal/tabla existentes. No usa
// el puerto serial.
func (a *App) CargarCalibracionSim(rutaCSV string) string {
	if a.estadoActual != EstadoDetenido {
		return "Error: detenga el sistema antes de cargar una calibración."
	}
	if rutaCSV == "" {
		return "No se seleccionó ningún archivo de calibración."
	}
	datos, err := leerCSVcrudo(rutaCSV)
	if err != nil {
		return "Error al leer el CSV de calibración: " + err.Error()
	}
	// El protocolo usa ventanas hasta [21,29) s; exigimos casi los 30 s completos.
	minMuestras := 29 * config.Fs
	if len(datos) < minMuestras {
		return fmt.Sprintf("Error: el CSV tiene %d muestras (%.1f s). La calibración 10-10-10 necesita ~30 s (mínimo %.0f s).",
			len(datos), float64(len(datos))/float64(config.Fs), float64(minMuestras)/float64(config.Fs))
	}

	res, pend := a.procesarEntrenamiento(datos)
	a.recordMutex.Lock()
	a.entrenamientoResult = res
	a.entPend = pend
	a.simCalibLista = true
	a.recordMutex.Unlock()

	runtime.EventsEmit(a.ctx, "entrenamiento-result", res)
	return fmt.Sprintf("Calibración cargada: %d muestras (%.1f s) procesadas.",
		len(datos), float64(len(datos))/float64(config.Fs))
}

// IniciarSimulacion reproduce un CSV de tiempo libre por el pipeline usando la base
// fija/umbrales de la calibración. Si puerto != "" intenta abrir el ESP32 (firmware
// de simulación) para mover los motores físicos; si falla, la simulación continúa
// SOLO en pantalla (los motores son opcionales).
func (a *App) IniciarSimulacion(rutaCSV string, puerto string) string {
	if a.estadoActual != EstadoDetenido {
		return "Error: el sistema ya está en ejecución."
	}
	if !a.baselineEntrenado {
		return "Error: confirma primero la calibración 10-10-10."
	}
	if rutaCSV == "" {
		return "No se seleccionó ningún archivo de tiempo libre."
	}
	datos, err := leerCSVcrudo(rutaCSV)
	if err != nil {
		return "Error al leer el CSV de tiempo libre: " + err.Error()
	}

	a.estadoActual = EstadoSimulando
	a.reiniciarEstado()

	// Fijar la base entrenada como referencia (igual que IniciarStreaming).
	a.paramsMu.Lock()
	a.C3_reposo_prom_mu = a.baseFijaC3Mu
	a.C3_reposo_prom_beta = a.baseFijaC3Beta
	a.C4_reposo_prom_mu = a.baseFijaC4Mu
	a.C4_reposo_prom_beta = a.baseFijaC4Beta
	a.paramsMu.Unlock()

	avisoPuerto := "solo en pantalla (sin ESP32)"
	if puerto != "" {
		port, perr := a.abrirPuertoSim(puerto)
		if perr != nil {
			a.debugLog("IniciarSimulacion: no se pudo abrir %s (%v); continúo solo en pantalla", puerto, perr)
		} else {
			a.serialMu.Lock()
			a.serialPort = port
			a.serialMu.Unlock()
			a.enviarComando('I', 0) // arrancar con la prótesis en reposo
			avisoPuerto = "con motores físicos en " + puerto
		}
	}

	ctx, cancel := context.WithCancel(context.Background())
	a.cancelRead = cancel
	go a.simularDatos(ctx, datos)

	return fmt.Sprintf("Simulación iniciada (%d muestras, %.1f s) %s.",
		len(datos), float64(len(datos))/float64(config.Fs), avisoPuerto)
}

// DetenerSimulacion cancela la reproducción, vuelve el ESP32 a reposo y cierra el
// puerto si estaba abierto.
func (a *App) DetenerSimulacion() string {
	if a.estadoActual != EstadoSimulando {
		return "No hay una simulación activa."
	}
	a.estadoActual = EstadoDetenido
	a.enviarComando('I', 0)
	if a.cancelRead != nil {
		a.cancelRead()
	}
	a.serialMu.Lock()
	if a.serialPort != nil {
		a.serialPort.Close()
		a.serialPort = nil
	}
	a.serialMu.Unlock()
	return "Simulación detenida."
}

// simularDatos reproduce las muestras de un CSV a ~tiempo real (config.Fs Hz),
// pasándolas por procesarMuestra (la misma cadena DSP que el modo en vivo). Al
// terminar el archivo vuelve a EstadoDetenido, cierra el puerto y emite "sim-fin".
func (a *App) simularDatos(ctx context.Context, datos [][]int32) {
	d := a.derived
	tick := time.NewTicker(time.Second / time.Duration(config.Fs))
	defer tick.Stop()

	for _, fila := range datos {
		select {
		case <-ctx.Done():
			return // DetenerSimulacion ya se encarga de cerrar el puerto
		case <-tick.C:
		}
		a.procesarMuestra(fila, d)
	}

	// Fin natural del CSV: prótesis a reposo, cerrar puerto y avisar al frontend.
	a.enviarComando('I', 0)
	a.serialMu.Lock()
	if a.serialPort != nil {
		a.serialPort.Close()
		a.serialPort = nil
	}
	a.serialMu.Unlock()
	a.estadoActual = EstadoDetenido
	runtime.EventsEmit(a.ctx, "sim-fin", len(datos))
}

// capturarEntrenamiento graba 30 s emitiendo cues de fase y, al terminar,
// calcula bases y umbrales automáticos.
func (a *App) capturarEntrenamiento(ctx context.Context, port serial.Port) {
	const nObjetivo = 30 * config.Fs // 7500 muestras

	select {
	case <-ctx.Done():
		return
	case <-time.After(800 * time.Millisecond):
	}
	port.ResetInputBuffer()

	a.recordMutex.Lock()
	a.recordedData = make([][]int32, 0, nObjetivo)
	a.recordMutex.Unlock()

	runtime.EventsEmit(a.ctx, "entrenamiento-progress", 0, "REPOSO")

	payload := make([]byte, 32)
	oneByte := make([]byte, 1)
	capturadas := 0

	for capturadas < nObjetivo {
		// leerExacto revisa ctx.Done() y tolera timeouts; devuelve error si el
		// puerto se cierra o se cancela ctx (Repetir/Confirmar/Detener).
		if err := a.leerExacto(ctx, port, oneByte, time.Time{}); err != nil {
			return
		}
		if oneByte[0] != 0xA0 {
			continue
		}
		if err := a.leerExacto(ctx, port, oneByte, time.Time{}); err != nil {
			return
		}
		if oneByte[0] != 0xC0 {
			continue
		}
		if err := a.leerExacto(ctx, port, payload, time.Time{}); err != nil {
			return
		}
		fila := make([]int32, 8)
		for i := 0; i < 8; i++ {
			fila[i] = int32(binary.LittleEndian.Uint32(payload[i*4 : (i+1)*4]))
		}
		a.recordMutex.Lock()
		a.recordedData = append(a.recordedData, fila)
		a.recordMutex.Unlock()

		capturadas++
		if capturadas%config.Fs == 0 {
			seg := capturadas / config.Fs
			runtime.EventsEmit(a.ctx, "entrenamiento-progress", seg, faseEntrenamiento(seg))
		}
	}

	a.recordMutex.Lock()
	datos := make([][]int32, len(a.recordedData))
	copy(datos, a.recordedData)
	a.recordMutex.Unlock()

	res, pend := a.procesarEntrenamiento(datos)

	a.recordMutex.Lock()
	a.entrenamientoResult = res
	a.entPend = pend
	a.recordMutex.Unlock()

	runtime.EventsEmit(a.ctx, "entrenamiento-result", res)
}

// faseEntrenamiento devuelve la fase (cue) según los segundos transcurridos.
func faseEntrenamiento(seg int) string {
	if seg >= 10 && seg < 20 {
		return "MOVIMIENTO"
	}
	return "REPOSO"
}

// procesarEntrenamiento corre el pipeline online de Go sobre el buffer grabado y
// calcula las bases (reposos) y los umbrales automáticos (fracción del ERD).
func (a *App) procesarEntrenamiento(datos [][]int32) (models.EntrenamientoResult, pendienteEntrenamiento) {
	a.paramsMu.RLock()
	p := a.params
	a.paramsMu.RUnlock()
	d := config.Derive(p)

	// Filtros locales (mismo pipeline que leerDatos) para no tocar los del realtime.
	notchSOS, notchGain := dsp.GetFilterNotch60(p.Orden_notch)
	nC3 := dsp.NewIIRFilter(notchSOS, notchGain)
	nC4 := dsp.NewIIRFilter(notchSOS, notchGain)
	var hpC3, hpC4 *dsp.IIRFilter
	if p.Eliminar_ruido_mov_resp {
		hpSOS, hpGain := dsp.GetFilterHP(p.Orden_hp)
		hpC3 = dsp.NewIIRFilter(hpSOS, hpGain)
		hpC4 = dsp.NewIIRFilter(hpSOS, hpGain)
	}
	muSOS, muGain := dsp.GetFilterMu(p.Orden_mu)
	c3muF := dsp.NewIIRFilter(muSOS, muGain)
	c4muF := dsp.NewIIRFilter(muSOS, muGain)
	betaSOS, betaGain := dsp.GetFilterBeta(p.Orden_beta)
	c3betaF := dsp.NewIIRFilter(betaSOS, betaGain)
	c4betaF := dsp.NewIIRFilter(betaSOS, betaGain)

	medC3mu := dsp.NewMedianFilter(d.N_median)
	medC4mu := dsp.NewMedianFilter(d.N_median)
	medC3beta := dsp.NewMedianFilter(d.N_median)
	medC4beta := dsp.NewMedianFilter(d.N_median)
	emaC3mu := dsp.NewEMAFilter(p.Alpha)
	emaC4mu := dsp.NewEMAFilter(p.Alpha)
	emaC3beta := dsp.NewEMAFilter(p.Alpha)
	emaC4beta := dsp.NewEMAFilter(p.Alpha)

	n := len(datos)
	pC3mu := make([]float64, n)
	pC4mu := make([]float64, n)
	pC3beta := make([]float64, n)
	pC4beta := make([]float64, n)

	for i := 0; i < n; i++ {
		// Mapeo de canales (igual que leerDatos), escala con ScaleFactor (VRef=5.0)
		c3 := float64(datos[i][0]) * ScaleFactor
		c3p := float64(datos[i][1]) * ScaleFactor
		c4 := float64(datos[i][2]) * ScaleFactor
		c4p := float64(datos[i][3]) * ScaleFactor
		c3pp := float64(datos[i][6]) * ScaleFactor
		c4pp := float64(datos[i][7]) * ScaleFactor

		diffC3 := c3 - (c3p+c3pp)/2.0
		diffC4 := c4 - (c4p+c4pp)/2.0

		cleanC3 := nC3.ProcessSample(diffC3)
		cleanC4 := nC4.ProcessSample(diffC4)
		if hpC3 != nil {
			cleanC3 = hpC3.ProcessSample(cleanC3)
			cleanC4 = hpC4.ProcessSample(cleanC4)
		}
		fC3mu := c3muF.ProcessSample(cleanC3)
		fC4mu := c4muF.ProcessSample(cleanC4)
		fC3beta := c3betaF.ProcessSample(cleanC3)
		fC4beta := c4betaF.ProcessSample(cleanC4)

		pC3mu[i] = math.Log(emaC3mu.Process(medC3mu.Process(fC3mu*fC3mu)) + 1e-8)
		pC4mu[i] = math.Log(emaC4mu.Process(medC4mu.Process(fC4mu*fC4mu)) + 1e-8)
		pC3beta[i] = math.Log(emaC3beta.Process(medC3beta.Process(fC3beta*fC3beta)) + 1e-8)
		pC4beta[i] = math.Log(emaC4beta.Process(medC4beta.Process(fC4beta*fC4beta)) + 1e-8)
	}

	fs := float64(config.Fs)
	mediaRango := func(arr []float64, tIni, tFin float64) float64 {
		ini := int(tIni * fs)
		fin := int(tFin * fs)
		if ini < 0 {
			ini = 0
		}
		if fin > n {
			fin = n
		}
		if ini >= fin {
			return 0
		}
		return calcularMedia(arr[ini:fin])
	}
	// Reposo combinado [1,9) ∪ [21,29) (ventanas iguales => media de las dos medias)
	promReposo := func(arr []float64) float64 {
		return (mediaRango(arr, 1, 9) + mediaRango(arr, 21, 29)) / 2.0
	}

	// Bases (reposo)
	c3BaseMu := promReposo(pC3mu)
	c3BaseBeta := promReposo(pC3beta)
	c4BaseMu := promReposo(pC4mu)
	c4BaseBeta := promReposo(pC4beta)

	// Movimiento [11,19)
	movC3mu := mediaRango(pC3mu, 11, 19)
	movC4mu := mediaRango(pC4mu, 11, 19)
	movC3beta := mediaRango(pC3beta, 11, 19)
	movC4beta := mediaRango(pC4beta, 11, 19)

	// Rebote (ERS) [20,22)
	rebC3beta := mediaRango(pC3beta, 20, 22)
	rebC4beta := mediaRango(pC4beta, 20, 22)

	// ERD observado (log; negativo = caída). El que más cae (winner) define el umbral.
	erdMuObs := math.Min(movC3mu-c3BaseMu, movC4mu-c4BaseMu)
	erdBetaObs := math.Min(movC3beta-c3BaseBeta, movC4beta-c4BaseBeta)
	// ERS observado (log; positivo = rebote). El mayor rebote define el umbral.
	ersBetaObs := math.Max(rebC3beta-c3BaseBeta, rebC4beta-c4BaseBeta)

	// Umbrales = fracción del observado
	frac := p.K_fraccion_erd
	umbralErdMu := frac * erdMuObs
	umbralErdBeta := frac * erdBetaObs
	umbralErsBeta := frac * ersBetaObs

	// Sliders equivalentes (mismo mapeo que actualizarUmbralesDesdeSliders):
	// umbral = log(1 + slider/100)  =>  slider = (exp(umbral) - 1) * 100
	sliderErdMu := (math.Exp(umbralErdMu) - 1) * 100
	sliderErdBeta := (math.Exp(umbralErdBeta) - 1) * 100
	sliderErsBeta := (math.Exp(umbralErsBeta) - 1) * 100

	// k (proporciones) para mostrar/guardar
	kErdMu := 1 - math.Exp(umbralErdMu)
	kErdBeta := 1 - math.Exp(umbralErdBeta)
	kErsBeta := math.Exp(umbralErsBeta) - 1

	// Porcentajes observados para la tabla
	erdMuPct := (1 - math.Exp(erdMuObs)) * 100
	erdBetaPct := (1 - math.Exp(erdBetaObs)) * 100
	ersBetaPct := (math.Exp(ersBetaObs) - 1) * 100

	// Tabla por canal/banda
	filas := [][]string{}
	addRow := func(nombre string, base, mov float64) {
		erd := mov - base
		pct := (1 - math.Exp(erd)) * 100
		filas = append(filas, []string{
			nombre,
			fmt.Sprintf("%.4f", base),
			fmt.Sprintf("%.4f", mov),
			fmt.Sprintf("%+.1f%%", pct),
			veredictoERD(pct),
		})
	}
	addRow("C3 mu", c3BaseMu, movC3mu)
	addRow("C4 mu", c4BaseMu, movC4mu)
	addRow("C3 beta", c3BaseBeta, movC3beta)
	addRow("C4 beta", c4BaseBeta, movC4beta)

	res := models.EntrenamientoResult{
		C3BaseMu:   c3BaseMu,
		C3BaseBeta: c3BaseBeta,
		C4BaseMu:   c4BaseMu,
		C4BaseBeta: c4BaseBeta,
		ErdMuPct:   erdMuPct,
		ErdBetaPct: erdBetaPct,
		ErsBetaPct: ersBetaPct,
		KErdMu:     kErdMu,
		KErdBeta:   kErdBeta,
		KErsBeta:   kErsBeta,
		Filas:      filas,
	}
	pend := pendienteEntrenamiento{
		baseC3Mu:      c3BaseMu,
		baseC3Beta:    c3BaseBeta,
		baseC4Mu:      c4BaseMu,
		baseC4Beta:    c4BaseBeta,
		sliderErdMu:   sliderErdMu,
		sliderErdBeta: sliderErdBeta,
		sliderErsBeta: sliderErsBeta,
	}
	return res, pend
}

// veredictoERD clasifica el porcentaje de ERD (mismos cortes que protocolo.m).
func veredictoERD(pct float64) string {
	switch {
	case pct > 20:
		return "✓✓ ERD claro"
	case pct > 10:
		return "✓ ERD presente"
	case pct > 0:
		return "~ leve ERD"
	case pct > -10:
		return "~ leve ERS"
	default:
		return "⚠ ERS"
	}
}

		return "No hay datos de entrenamiento para guardar."

	sb.WriteString("-----------------------------------------\n")

	sb.WriteString(fmt.Sprintf("%-10s %12s %12s %10s   %s\n", "Canal", "Base(log)", "Mov(log)", "ERD%", "Veredicto"))
	for _, f := range res.Filas {
		sb.WriteString(fmt.Sprintf("%-10s %12s %12s %10s   %s\n", f[0], f[1], f[2], f[3], f[4]))
	}
	sb.WriteString("-----------------------------------------\n")
	sb.WriteString("UMBRALES AUTOMÁTICOS (fracción del ERD observado)\n")
	sb.WriteString(fmt.Sprintf("  k_erd_mu   = %.4f (%.1f%%)\n", res.KErdMu, res.KErdMu*100))
	sb.WriteString(fmt.Sprintf("  k_erd_beta = %.4f (%.1f%%)\n", res.KErdBeta, res.KErdBeta*100))

	sb.WriteString("=========================================\n")

	if err := os.WriteFile(rutaTXT, []byte(sb.String()), 0644); err != nil {
		return "CSV guardado, pero error al crear TXT: " + err.Error()
	}
	return fmt.Sprintf("Entrenamiento guardado: %d muestras en CSV y resumen en TXT.", len(datos))
}

func calcularMediaStd(datos []float64) (float64, float64) {
	if len(datos) == 0 {
		return 0, 0
	}
	sum := 0.0
	for _, v := range datos {
		sum += v
	}
	mean := sum / float64(len(datos))
	var varianza float64
	for _, v := range datos {
		varianza += (v - mean) * (v - mean)
	}
	return mean, math.Sqrt(varianza / float64(len(datos)))
}

func calcularMedia(datos []float64) float64 {
	if len(datos) == 0 {
		return 0
	}
	sum := 0.0
	for _, v := range datos {
		sum += v
	}
	return sum / float64(len(datos))
}

func calcularPearson(x, y []float64) float64 {
	if len(x) != len(y) || len(x) == 0 {
		return 0
	}
	meanX := calcularMedia(x)
	meanY := calcularMedia(y)
	var sumXY, sumX2, sumY2 float64
	for i := range x {
		dx := x[i] - meanX
		dy := y[i] - meanY
		sumXY += dx * dy
		sumX2 += dx * dx
		sumY2 += dy * dy
	}
	if sumX2 == 0 || sumY2 == 0 {
		return 0
	}
	return sumXY / math.Sqrt(sumX2*sumY2)
}

func max(a, b int) int {
	if a > b {
		return a
	}
	return b
}

    const logErsBeta = Math.log(1.0 + (this.thErsBetaValue() / 100.0));

    if (this.c4MuC)   this.dibujarCanvas(this.c4MuC.nativeElement,   [this.dC4Mu],              ['#fff'],              [logErdMu],              false, -2, 2);
    if (this.c3BetaC) this.dibujarCanvas(this.c3BetaC.nativeElement, [this.dC3B],               ['#fff'],              [logErdBeta, logErsBeta],false, -2, 2);
    if (this.c4BetaC) this.dibujarCanvas(this.c4BetaC.nativeElement, [this.dC4B],               ['#fff'],              [logErdBeta, logErsBeta],false, -2, 2);

    if (this.motorC)  this.dibujarCanvas(this.motorC.nativeElement,  [this.dMotC3, this.dMotC4],['#58a6ff', '#ff5555'],[],                      true,  -0.5, 1.5);

  dibujarCanvas(
    canvas: HTMLCanvasElement,
    datosArr: number[][], colores: string[],
    umbrales: number[], esPotencia: boolean,
    limiteMin: number | null, limiteMax: number | null
  ) {
    if (datosArr[0].length === 0) return;
    const ctx = canvas.getContext('2d')!;
    const w = canvas.width, h = canvas.height;

    ctx.fillStyle = '#0d1117';
    ctx.fillRect(0, 0, w, h);

    let min = Infinity, max = -Infinity;
    if (limiteMin !== null && limiteMax !== null) {
      min = limiteMin; max = limiteMax;
    } else {
      datosArr.forEach(datos => datos.forEach(v => { if (v < min) min = v; if (v > max) max = v; }));
      umbrales.forEach(u => { if (u && u < min) min = u; if (u && u > max) max = u; });
      if (min === max) { min -= 1; max += 1; }
      const margen = (max - min) * 0.15;
      min -= margen; max += margen;
      if (esPotencia && min > 0) min = 0;
    }

    const rango = max - min;
    const pasoX = w / this.maxMuestras;

    // Línea de cero
    if (min < 0 && max > 0) {
      const yZero = h - ((0 - min) / rango) * h;
      ctx.strokeStyle = '#30363d'; ctx.lineWidth = 1; ctx.beginPath();
      ctx.moveTo(0, yZero); ctx.lineTo(w, yZero); ctx.stroke();
    }

    // Umbrales punteados rojos
    ctx.setLineDash([5, 5]); ctx.strokeStyle = '#ff5555'; ctx.lineWidth = 1;
    umbrales.forEach(u => {
      if (u) {
        const yU = h - ((u - min) / rango) * h;
        ctx.beginPath(); ctx.moveTo(0, yU); ctx.lineTo(w, yU); ctx.stroke();
      }
    });
    ctx.setLineDash([]);

    // Señales
    datosArr.forEach((datos, idx) => {
      ctx.beginPath(); ctx.strokeStyle = colores[idx]; ctx.lineWidth = 1.5;
      datos.forEach((v, i) => {
        const x = i * pasoX;
        const y = h - ((v - min) / rango) * h;
        i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
      });
      ctx.stroke();
    });
  }
}
